package controller;

import javax.sound.sampled.*;
import java.io.IOException;

public class AudioManager {

    private static Clip bgmClip;
    private static FloatControl volumeControl;

    private static int currentVolume = 50;

    public static void playMusic(String filePath) {
        try {
            if (bgmClip != null && bgmClip.isRunning()) {
                return;
            }

            java.net.URL audioUrl = AudioManager.class.getResource(filePath);
            if (audioUrl == null) {
                System.err.println("No se encontrÃ³ el archivo de audio: " + filePath);
                return;
            }

            AudioInputStream audioStream = AudioSystem.getAudioInputStream(audioUrl);

            bgmClip = AudioSystem.getClip();
            bgmClip.open(audioStream);

            volumeControl = (FloatControl) bgmClip.getControl(FloatControl.Type.MASTER_GAIN);
            setVolume(currentVolume);

            bgmClip.loop(Clip.LOOP_CONTINUOUSLY);
            bgmClip.start();

        } catch (UnsupportedAudioFileException | IOException | LineUnavailableException e) {
            System.err.println("Error al reproducir audio: " + e.getMessage());
        }
    }

    public static void stopMusic() {
        if (bgmClip != null && bgmClip.isRunning()) {
            bgmClip.stop();
        }
    }

    public static void setVolume(int percentage) {
        if (volumeControl != null) {
            if (percentage <= 0) {
                volumeControl.setValue(volumeControl.getMinimum());
            } else {
                float dB = (float) (Math.log10((double) percentage / 100.0) * 20.0);
                volumeControl.setValue(dB);
            }
        }
    }
}

package controller;

import model.GameModel;
import view.GamePanel;

import javax.swing.*;

public class GameLoopThread extends Thread {

    private static final int msPorFrame = 16; // 60 FPS -- 1000/60=16.66
    private final GamePanel gamePanel;
    private final GameModel model;

    public GameLoopThread(GamePanel gamePanel, GameModel model) {
        this.gamePanel = gamePanel;
        this.model = model;
        setDaemon(true); // Hilo de fondo para apagar el programa entero
    }

    @Override
    public void run() {  // Se redefine lo que debe hacer el hilo cuando se ejecute el .start()
        while (true) { // Game Loop
            model.update();
            SwingUtilities.invokeLater(gamePanel :: repaint);
            try {
                Thread.sleep(msPorFrame); // Se pide al hilo que "descanse" por 16 ms para que el bucle no use el 100% del procesador
            } catch (InterruptedException e) {
                Thread.currentThread().interrupt(); // Maneja la excepciÃ³n por si se cierra cuando esta "descansando"
                break;
            }
        }
    }
}

package model;

public class Bullet extends Entity {
    private final double dirX;
    private final double dirY;
    private final double speed;
    private final double damage;
    private final BulletKind kind;
    private boolean active; // booleano para saber si la bala debe 'desaparecer'
    private int lifeSpanTicks;
    private int age;        // ticks que lleva viva (la vista lo usa para que la estela "salga" del arma)

    public Bullet(double x, double y, int width, int height, double dirX, double dirY, double damage, int lifeSpanTicks) {
        this(x, y, width, height, dirX, dirY, damage, lifeSpanTicks, BulletKind.STANDARD);
    }

    public Bullet(double x, double y, int width, int height, double dirX, double dirY, double damage, int lifeSpanTicks, BulletKind kind) {
        super(x, y, width, height);
        this.dirX = dirX;
        this.dirY = dirY;
        this.speed = 12.0;
        this.damage = damage;
        this.active = true;
        this.lifeSpanTicks = lifeSpanTicks;
        this.kind = kind;
        this.age = 0;
    }

    public void update(int mapWidth, int mapHeight) {
        this.x += dirX * speed; 
        this.y += dirY * speed;

        this.age++;
        this.lifeSpanTicks--;

        if (this.lifeSpanTicks <= 0) {
            this.active = false;
        }

        // Desactivar la bala si sale de los lÃ­mites de la ventana
        if (x < 0 || x > mapWidth || y < 0 || y > mapHeight) {
            active = false;
        }
    }

    public boolean isActive() {
        return active;
    }
    public void setActive(boolean active) {
        this.active = active;
    }
    public double getDamage(){
        return damage;
    }
    public double getDirX() {
        return dirX;
    }
    public double getDirY() {
        return dirY;
    }
    public BulletKind getKind() {
        return kind;
    }
    public int getAge() {
        return age;
    }
    public double getSpeed() {
        return speed;
    }
}

package model;

public abstract class Character extends Entity {

    protected double speed;
    private int health;
    private int maxHealth;

    public Character(double x, double y, int width, int height, double speed, int health, int maxHealth) {
        super(x, y, width, height);
        this.speed = speed;
        this.maxHealth = maxHealth;
        this.health = health;
    }

    public int getHealth() {
        return health;
    }
    public void setHealth(int health) {
        this.health = health;
    }

    public int getMaxHealth() {
        return maxHealth;
    }

    protected void move(double dx, double dy, GameState state) {
        if (dx != 0 || dy != 0) {
            double length = Math.sqrt(dx * dx + dy * dy);
            double moveX = (dx / length) * speed;
            double moveY = (dy / length) * speed;

            if (!isCollidingWithMap(x + moveX, y, state)) x += moveX;
            if (!isCollidingWithMap(x, y + moveY, state)) y += moveY;
        }
    }

    protected boolean isCollidingWithMap(double nextX, double nextY, GameState state) {
        boolean[][] map = state.getCollisionMatrix();
        if (map == null) return false;

        int tile_size = GameState.TILE_SIZE;

        // Hitbox lÃ³gica (mÃ¡s chica que sprite visual)
        int leftTile = (int) (nextX + 15) / tile_size;
        int rightTile = (int) (nextX + width - 15) / tile_size;
        int topTile = (int) (nextY + 30) / tile_size;
        int bottomTile = (int) (nextY + height - 5) / tile_size;

        if (leftTile < 0 || rightTile >= map.length || topTile < 0 || bottomTile >= map[0].length) {
            return true;
        }

        return map[leftTile][topTile] || map[rightTile][topTile] ||
           map[leftTile][bottomTile] || map[rightTile][bottomTile];
    }

    protected boolean canMove(double dx, double dy, GameState state) {
        if (dx == 0 && dy == 0) return false;

        double length = Math.hypot(dx, dy);
        double moveX = (dx / length) * speed;
        double moveY = (dy / length) * speed;

        return !isCollidingWithMap(x + moveX, y + moveY, state);
    }

        return speed;
    }

    public void setSpeed(double speed) {
        this.speed = speed;
    }

    public void takeDamage(int amount) {
        if (amount < this.health) {
            health -= amount;
        } else {
            this.health = 0;
        }
    }

    public boolean isAlive() {
        return health > 0;
    }

public class CollisionSystem {

    public void update (GameState state) {
        checkBulletZombieCollision(state);
        checkBulletMapCollision(state);
        checkPoisonPlayerCollision(state);
        checkZombiePlayerCollision(state);
    }

    private void checkBulletZombieCollision (GameState state) {
        for (Bullet bullet : state.getPlayerBullets()) {
            if (!bullet.isActive()) continue;

            for (Enemy zombie : state.getEnemies()) {
                if (zombie.isAlive() && bullet.collidesWith(zombie)) {
                    bullet.setActive(false);

                    int healthBefore = zombie.getHealth();
                    zombie.takeDamage((int) bullet.getDamage());
                    state.getMatchStats().addDamageDealt(healthBefore - zombie.getHealth());
                    break;
                }
            }
        }
    }

    private void checkBulletMapCollision (GameState state) {
        boolean[][] map = state.getCollisionMatrix();
        if (map == null) return;

        int tileSize = GameState.TILE_SIZE;

        for (Bullet bullet : state.getPlayerBullets()) {
            if (!bullet.isActive()) continue;

            int tileX = (int) (bullet.getX() + bullet.getWidth() / 2.0) / tileSize;
            int tileY = (int) (bullet.getY() + bullet.getHeight() / 2.0) / tileSize;

            if (tileX >= 0 && tileX < map.length && tileY >= 0 && tileY < map[0].length) {
                if (map[tileX][tileY]) {
                    bullet.setActive(false);
                }
            }
        }

        for (PoisonBullet poison : state.getPoisonBullets()) {
            if (!poison.isActive()) continue;

            int tileX = (int) (poison.getX() + poison.getWidth() / 2.0) / tileSize;
            int tileY = (int) (poison.getY() + poison.getHeight() / 2.0) / tileSize;

            if (tileX >= 0 && tileX < map.length && tileY >= 0 && tileY < map[0].length) {
                if (map[tileX][tileY]) {
                    poison.setActive(false);
                }
            }
        }
    }

    private void checkPoisonPlayerCollision (GameState state) {
        Player player = state.getPlayer();
        if (!player.isAlive()) return;

        for (PoisonBullet poison : state.getPoisonBullets()) {
            if (poison.isActive() && poison.collidesWith(player)) {
                poison.setActive(false);
                poison.applyTo(player);//POLIMORFISMO: cada bala sabe quÃ© hacer
            }
        }
    }

    private void checkZombiePlayerCollision(GameState state) {
        Player player = state.getPlayer();
        if (!player.isAlive()) return;

        for (Enemy zombie : state.getEnemies()) {
            if (!zombie.isAlive()) continue;

            if (zombie.collidesWith(player)) {
                // 1. Si es el primer instante en que choca, inicia el temporizador del choque
                if (!zombie.isCollidingWithPlayer()) {
                    zombie.setCollidingWithPlayer(true);
                    zombie.setCollisionStartTime(System.currentTimeMillis());
                }

                // 2. Debe tener su intervalo de ataque listo Y haber mantenido el choque por 400ms
                if (zombie.canAttack() && zombie.hasCompletedAttackDelay()) {
                    player.takeDamage(zombie.attack());
                    zombie.resetAttackCooldown();
                }
            } else {
                // 3. Si el jugador se aleja antes de los 400ms, se reinicia el choque
                zombie.setCollidingWithPlayer(false);
            }
        }
    }
}

package model;

public abstract class Enemy extends Character {

    private final int ATTACK_COOLDOWN= 1000;
    protected double angle;
    private long lastAttackTime = 0;
    private static final double[] AVOID_ANGLES_DEG = {0, 25, -25, 50, -50, 75, -75, 100, -100};
    private boolean isCollidingWithPlayer = false;
    private long collisionStartTime = 0;
    private final int ATTACK_WINDUP = 400;

    public Enemy(double x, double y, int width, int height, double speed, int health, int maxHealth) {
        super(x, y, width, height, speed, health, maxHealth);
        this.angle = 0.0;
    }

    public void setX(double x) {
        this.x = x;
    }

    public void setY(double y) {
        this.y = y;
    }

    public double getAngle() {
        return angle;
    }

    public boolean isCollidingWithPlayer() {
        return isCollidingWithPlayer;
    }

    public void setCollidingWithPlayer(boolean colliding) {
        this.isCollidingWithPlayer = colliding;
    }

    public void setCollisionStartTime(long time) {
        this.collisionStartTime = time;
    }

    public boolean hasCompletedAttackDelay() {
        return (System.currentTimeMillis() - collisionStartTime >= ATTACK_WINDUP);
    }

    public void update(Player target, GameState state) {
        // Verifica si el zombie estÃ¡ muerto antes de moverse
        if (!isAlive()) return;

        double playerCenterX = target.getX() + (target.getWidth() / 2.0);
        double playerCenterY = target.getY() + (target.getHeight() / 2.0);
        double zombieCenterX = this.x + (this.width / 2.0);
        double zombieCenterY = this.y + (this.height / 2.0);

        double dx = playerCenterX - zombieCenterX;
        double dy = playerCenterY - zombieCenterY;

        double distance = Math.hypot(dx, dy);
        this.angle = Math.atan2(dy, dx);

        if (distance > 1000) {
            return;
        }

        if (distance > 30) {
            moveTowardsWithAvoidance(dx, dy, state);
        }

        avoidOverlap(state);
    }

    private void moveTowardsWithAvoidance(double dx, double dy, GameState state) {
        double baseAngle = Math.atan2(dy, dx);
        double length = Math.hypot(dx, dy);

        if (length == 0) return;

        for (double offsetDeg : AVOID_ANGLES_DEG) {
            double testAngle = baseAngle + Math.toRadians(offsetDeg);
            double testDx = Math.cos(testAngle) * length;
            double testDy = Math.sin(testAngle) * length;

            if (canMove(testDx, testDy, state)) {
                move(testDx, testDy, state);
                return;
            }
        }
    }

    public void avoidOverlap(GameState state) {
        for (Enemy otherZombie : state.getEnemies()) {
            if (this == otherZombie || !otherZombie.isAlive()) continue;

            if (this.collidesWith(otherZombie)) {

                double dx = this.getX() - otherZombie.getX();
                double dy = this.getY() - otherZombie.getY();
                double distance = Math.sqrt(dx * dx + dy * dy);

                if (distance > 0) {
                    double pushX = (dx/distance) * 1.5;
                    double pushY = (dy/distance) * 1.5;

                    if (!isCollidingWithMap(x + pushX, y, state)) {
                        x += pushX;
                    }

                    if (!isCollidingWithMap(x, y + pushY, state)) {
                        y += pushY;
                    }
                }
            }
        }
    }

    public boolean canAttack() {
        long currentTime = System.currentTimeMillis();
        return (currentTime - lastAttackTime >= ATTACK_COOLDOWN);
    }

    public void resetAttackCooldown() {
        this.lastAttackTime = System.currentTimeMillis();
    }

    public abstract int attack();

    public boolean isAttacking() {
        return (System.currentTimeMillis() - lastAttackTime < 500);
    }
}

import java.util.Random;

public class EnemySpawner {
    private long lastSpawnTime;
    private int spawnInterval;
    private int zombieSpawned;
    private final int waveLimit;

    public EnemySpawner(int spawnInterval, int waveLimit){
        this.spawnInterval = spawnInterval;
        this.waveLimit = waveLimit;
        this.lastSpawnTime = System.currentTimeMillis();
        this.zombieSpawned = 0;
    }
    
    public void update(GameState state){
        long currentTime = System.currentTimeMillis();

        if (currentTime - lastSpawnTime >= spawnInterval && zombieSpawned < waveLimit) {
            Random random = new Random();
            int tipoZombie = random.nextInt(4);

            int mapWidth = state.getMapWidth();
            int mapHeight = state.getMapHeight();
            boolean[][] collisionMatrix = state.getCollisionMatrix();

            if (mapWidth <= 100 || mapHeight <= 100 || collisionMatrix == null) return;

            int spawnX = 0;
            int spawnY = 0;
            boolean validPosition = false;
            int maxAttempts = 50; 
            int attempts = 0;
            int maxZombieWidth = 86;
            int maxZombieHeight = 86;

            while (!validPosition && attempts < maxAttempts) {
                spawnX = random.nextInt(mapWidth - maxZombieWidth);
                spawnY = random.nextInt(mapHeight - maxZombieHeight);
                if (!isOnBlockedTile(spawnX, spawnY, maxZombieWidth, maxZombieHeight, collisionMatrix)) {
                    validPosition = true;
                }
                attempts++;
            }

            Enemy newZombie;

            if (tipoZombie == 0) {
                newZombie = new TankZombie(spawnX, spawnY);
            } else if (tipoZombie == 1) {
                newZombie = new FastZombie(spawnX, spawnY);
            } else if (tipoZombie == 2) {
                newZombie = new PoisonZombie(spawnX, spawnY, state.getPoisonBullets());
            } else {
                newZombie = new NormalZombie(spawnX, spawnY);
            }

            state.addEnemy(newZombie);
            lastSpawnTime = currentTime;
            zombieSpawned++;
        }
    }

    // MÃ©todo extraÃ­do de la lÃ³gica del PowerUpSpawner
    private boolean isOnBlockedTile(double x, double y, int width, int height, boolean[][] map) {
        int tile = GameState.TILE_SIZE;
        int fromX = (int) x / tile;
        int toX = (int) (x + width - 1) / tile;
        int fromY = (int) y / tile;
        int toY = (int) (y + height - 1) / tile;

        for (int tx = fromX; tx <= toX; tx++) {
            for (int ty = fromY; ty <= toY; ty++) {
                if (tx < 0 || tx >= map.length || ty < 0 || ty >= map[0].length) return true;
                if (map[tx][ty]) return true; // Hay colisiÃ³n
            }
        }
        return false; // Ãrea completamente libre
    }

    public boolean isWaveComplete(){
        return zombieSpawned >= waveLimit;
    }

    public void reset() {
        this.zombieSpawned = 0; 
        this.lastSpawnTime = System.currentTimeMillis();
    }
}

package model;

import java.time.LocalDate;
import java.util.List;

public class GameModel {

    private final GameState state;
    private final CollisionSystem collisionSystem;
    private final EnemySpawner spawner;
    private final PowerUpSpawner powerUpSpawner;

    public enum GameStatus {PLAYING, PAUSED, BOSS_INTRO, GAME_OVER, GAME_WON}
    private GameStatus status = GameStatus.PLAYING;
    private boolean bossSpawned = false;

    // Intro del jefe: 2 mensajes de INTRO_STEP_MS cada uno
    private static final int INTRO_STEPS = 2;
    private static final long INTRO_STEP_MS = 3500;
    private long bossIntroStart;

    private final MatchStats stats;
    private long lastUpdateTime = System.currentTimeMillis();

    private final MatchDAO matchDAO;

    private long matchTimeMs;
    private long bestTimeMs;
    private boolean newRecord;

    public GameModel(GameState state, MatchDAO matchDAO) {
        this.state = state;
        this.matchDAO = matchDAO;
        this.collisionSystem = new CollisionSystem();
        this.stats = state.getMatchStats();

        // Inicializamos los spawners
        this.spawner = new EnemySpawner(2000, 15);
        this.powerUpSpawner = new PowerUpSpawner(state, 4000, 20, 16); // cada 4 s, mÃ¡x. 20 en el mapa, 16 al empezar

        //METODO PARA SPAWNEAR EL BOSS EN EL ESTACIONAMIENTO
        //spawnBoss(6027, 5056);
    }

    public void update() {
        long now = System.currentTimeMillis();
        long delta = now - lastUpdateTime;
        lastUpdateTime = now;

        if (status == GameStatus.BOSS_INTRO) {
            updateBossIntro();
            return;
        }
        if (status != GameStatus.PLAYING) return;

        stats.addPlayedTime(delta);

        // Mover jugador
        Player player = state.getPlayer();
        if (!player.isAlive()) {
            saveMatch(false);
            status = GameStatus.GAME_OVER;
            return;
        }

        player.update(state);

        int mapWidth = state.getMapWidth();
        int mapHeight = state.getMapHeight();

        // Mover balas
        for (Bullet b : state.getPlayerBullets()) b.update(mapWidth, mapHeight);
        for (Bullet p: state.getPoisonBullets()) p.update(mapWidth, mapHeight);

        // Spawners
        spawner.update(state);
        powerUpSpawner.update();

        // Mover zombies
        for (Enemy zombie : state.getEnemies()) {
            if (zombie.isAlive()) {
                zombie.update(player, state);
            }
        }

        // RecolecciÃ³n de power-ups
        for (PowerUp p : state.getPowerUps()) {
            if (p.isActive() && player.collidesWith(p)) {
                p.collect(player);
            }
        }

        // Sistema de colisiones (daÃ±o)
        collisionSystem.update(state);

        // Los zombies muertos en este tick cuentan como eliminados
        for (Enemy zombie : state.getEnemies()) {
            if (!zombie.isAlive()) {
                stats.addKill();
            }
        }

        // Limpia los inactivos
        state.purgeInactive();

        // Oleada terminada -> intro del jefe; jefe derrotado -> fin de nivel
        if (state.getEnemies().isEmpty() && spawner.isWaveComplete()) {
            if (!bossSpawned) {
                startBossIntro();
            } else {
                saveMatch(true);
                status = GameStatus.GAME_WON;
            }
        }
    }

    private void startBossIntro() {
        status = GameStatus.BOSS_INTRO;
        bossIntroStart = System.currentTimeMillis();
    }

    private void updateBossIntro() {
        long elapsed = System.currentTimeMillis() - bossIntroStart;
        if (elapsed >= INTRO_STEP_MS * INTRO_STEPS) {
            spawnBoss(6027, 5056);
            status = GameStatus.PLAYING;
        }
    }

    private void spawnBoss(double x, double y) {
        state.addEnemy(new FinalBoss(x, y, state.getPoisonBullets()));
        bossSpawned = true;
    }

    /** QuÃ© mensaje toca mostrar: 0 = felicitaciÃ³n, 1 = "Preparate para el Jefe". */
    public int getBossIntroStep() {
        long elapsed = System.currentTimeMillis() - bossIntroStart;
        return (int) Math.min(INTRO_STEPS - 1, elapsed / INTRO_STEP_MS);
    }

    /** Avance (0.0 a 1.0) dentro del mensaje actual, para el efecto de aparecer y desvanecer. */
    public double getBossIntroProgress() {
        long elapsed = System.currentTimeMillis() - bossIntroStart;
        return Math.min(1.0, (elapsed % INTRO_STEP_MS) / (double) INTRO_STEP_MS);
    }

    public void togglePause() {
        if (status == GameStatus.PLAYING) {
            status = GameStatus.PAUSED;
        } else if (status == GameStatus.PAUSED) {
            status = GameStatus.PLAYING;
        }
    }

    public void playerInteract() {
        if (status != GameStatus.PLAYING) return;

        List<WeaponItem> groundWeapons = state.getGroundWeapons();
        Player player = state.getPlayer();

        for (int i = 0; i < groundWeapons.size(); i++) {
            WeaponItem item = groundWeapons.get(i);

            if (player.collidesWith(item)) {
                FireArm oldWeapon = player.getPrimaryWeapon();
                player.equipWeapon(item.getWeapon());
                groundWeapons.remove(i);

                if (oldWeapon != null) {
                    WeaponItem droppedWeapon = new WeaponItem(player.getX(), player.getY(), oldWeapon);
                    groundWeapons.add(droppedWeapon);
                }
                break;
            }
        }
    }

    public void saveMatch(boolean victory) {
        MatchRecord record = new MatchRecord(LocalDate.now(), stats.getPlayedTimes(), stats.getZombiesKilled(), stats.getDamageDealt(), state.getPlayer().getDamageTaken(), victory);



        this.matchTimeMs = record.getSurvivalTimeMs();

        if (victory) {
            List<MatchRecord> previousBest = matchDAO.fastestVictories(1);
            boolean noPrevious = previousBest.isEmpty();
            long previousBestMs = noPrevious ? matchTimeMs : previousBest.get(0).getSurvivalTimeMs();
            this.newRecord = noPrevious || matchTimeMs < previousBestMs;
            this.bestTimeMs = newRecord ? matchTimeMs : previousBestMs;

        }

        matchDAO.save(record);
    }

    public long getMatchTimeMs() {
        return matchTimeMs;
    }
    public long getBestTimeMs() {

package model;

import java.io.InputStream;
import java.util.Scanner;

public class MapLoader {

    public static boolean[][] loadCollisionMatrix(String filePath, int cols, int rows) {
        boolean[][] matrix = new boolean[cols][rows];
        try (InputStream is = MapLoader.class.getResourceAsStream(filePath)) {
            if (is == null) {
                System.err.println("No se encontrÃ³ el mapa en: " + filePath);
                return matrix;
            }
            Scanner scanner = new Scanner(is, "UTF-8").useDelimiter("\\A");
            String json = scanner.hasNext() ? scanner.next() : "";
            scanner.close();

            int startData = json.indexOf("\"data\":[");
            if (startData == -1) {
                System.err.println("ERROR MapLoader: No se encontrÃ³ ningÃºn array 'data'");
                return matrix;
            }
            int startBracket = json.indexOf("[", startData);
            int endBracket = json.indexOf("]", startBracket);

            if (startBracket == -1 || endBracket == -1) {
                System.err.println("ERROR MapLoader: No se encontraron los corchetes de data");
                return matrix;
            }

            String dataString = json.substring(startBracket + 1, endBracket);
            String[] values = dataString.split(",");

            System.out.println("MapLoader: Total de tiles leÃ­dos en la capa = " + values.length);

            int solidCount = 0;
            int col = 0;
            int row = 0;

            for (String val : values) {
                String trimmed = val.trim();
                if (!trimmed.isEmpty()) {
                    long id = Long.parseLong(trimmed);
                    boolean isSolid = (id > 0);

                    if (col < cols && row < rows) {
                        matrix[col][row] = isSolid;
                        if (isSolid) solidCount++;
                    }

                    col++;
                    if (col >= cols) {
                        col = 0;
                        row++;
                    }
                }
            }

            System.out.println("MapLoader: Carga completada. ObstÃ¡culos sÃ³lidos detectados: " + solidCount);
        } catch (Exception e) {
            System.err.println("Error leyendo el mapa: " + e.getMessage());
            e.printStackTrace();
        }

        return matrix;
    }
}

public class MeleeWeapon extends Weapon {

    private final double range = 100.0; // PÃ­xeles de alcance del cuchillo
    private final double arcAngle = Math.toRadians(90); // 90 grados de arco de ataque

    public MeleeWeapon(int damage, long cooldown) {
        super(damage);
        this.cooldown = cooldown;
    }

    @Override
    public boolean attack(Player player, GameState state) {
        long currentTime = System.currentTimeMillis();

        if (currentTime - this.lastShoot >= this.cooldown) {
            this.lastShoot = currentTime;

            double totalDamage = this.getDamage() + player.getDamageBonus();
            boolean hitSomething = false;

            for (Enemy zombie : state.getEnemies()) {
                if (!zombie.isAlive()) continue;

                double playerCenterX = player.getX() + (player.getWidth() / 2.0);
                double playerCenterY = player.getY() + (player.getHeight() / 2.0);

                double zombieCenterX = zombie.getX() + (zombie.getWidth() / 2.0);
                double zombieCenterY = zombie.getY() + (zombie.getHeight() / 2.0);

                double dx = zombieCenterX - playerCenterX;
                double dy = zombieCenterY - playerCenterY;

                double distance = Math.hypot(dx, dy);

                if (distance <= range) {
                    double angleToZombie = Math.atan2(dy, dx); // Calculo el Ã¡ngulo hacia el zombie

                    double angleDiff = angleToZombie - player.getAngle();
                    angleDiff = Math.atan2(Math.sin(angleDiff), Math.cos(angleDiff));

                    if (Math.abs(angleDiff) <= this.arcAngle / 2.0) {
                        int healthBefore = zombie.getHealth();
                        zombie.takeDamage((int) totalDamage);
                        state.getMatchStats().addDamageDealt(healthBefore - zombie.getHealth());
                        hitSomething = true;
                    }
                }
            }
            return hitSomething;
        }
        return false;
    }

    @Override
    public boolean isAutomatic() {
        return false;
    }
}

    private final long spawnInterval; // ms entre cada reposiciÃ³n
    private final int maxOnMap;       // tope de power ups simultÃ¡neos
    private long lastSpawnTime;
    private final Random random = new Random();

    public PowerUpSpawner(GameState state, long spawnInterval, int maxOnMap, int initialCount) {
        this.state = state;
        // La matriz de colisiones y la posiciÃ³n inicial del jugador ya tienen que estar cargadas acÃ¡
        this.reachableArea = new ReachableArea(state);
        if (!reachableArea.isUsable()) {
            System.err.println("PowerUpSpawner: no se pudo calcular la zona alcanzable; se usa solo la matriz de colisiones.");
        }
        this.powerUps = state.getPowerUps();
        this.spawnInterval = spawnInterval;
        this.maxOnMap = maxOnMap;
        this.lastSpawnTime = System.currentTimeMillis();
        populate(Math.min(initialCount, maxOnMap));
    }

    /** Reparte la cantidad inicial recorriendo los tipos en orden, para que haya variedad. */
    private void populate(int count) {
        int firstType = random.nextInt(TYPES);
        for (int i = 0; i < count; i++) {
            trySpawn((firstType + i) % TYPES);
        }
    }

    public void update() {
        long now = System.currentTimeMillis();
        if (now - lastSpawnTime < spawnInterval || powerUps.size() >= maxOnMap) return;
        lastSpawnTime = now;
        trySpawn(random.nextInt(TYPES));
    }

    private boolean trySpawn(int type) {
        int mapWidth = state.getMapWidth();
        int mapHeight = state.getMapHeight();
        if (mapWidth <= PowerUp.SIZE || mapHeight <= PowerUp.SIZE) return false;

        for (int attempt = 0; attempt < MAX_ATTEMPTS; attempt++) {
            double x = random.nextInt(mapWidth - PowerUp.SIZE);
            double y = random.nextInt(mapHeight - PowerUp.SIZE);
            if (isValidSpot(x, y)) {
                powerUps.add(create(type, x, y));
                return true;
            }
        }
        return false; // no encontrÃ³ lugar: lo intenta en la prÃ³xima reposiciÃ³n
    }

    private boolean isValidSpot(double x, double y) {
        if (isOnBlockedTile(x, y)) return false;

        double centerX = x + PowerUp.SIZE / 2.0;
        double centerY = y + PowerUp.SIZE / 2.0;

        Player player = state.getPlayer();
        if (reachableArea.isUsable() && !reachableArea.canPickUpAt(centerX, centerY, player)) return false;

        double playerX = player.getX() + player.getWidth() / 2.0;
        double playerY = player.getY() + player.getHeight() / 2.0;
        if (Math.hypot(centerX - playerX, centerY - playerY) < MIN_DIST_TO_PLAYER) return false;

        for (PowerUp other : powerUps) {
            double otherX = other.getX() + PowerUp.SIZE / 2.0;
            double otherY = other.getY() + PowerUp.SIZE / 2.0;
            if (Math.hypot(centerX - otherX, centerY - otherY) < MIN_DIST_BETWEEN) return false;
        }
        return true;
    }

    /** true si el power up toca algÃºn tile bloqueado del mapa o se sale de Ã©l. */
    private boolean isOnBlockedTile(double x, double y) {
        boolean[][] map = state.getCollisionMatrix();
        if (map == null) return false;

        int tile = GameState.TILE_SIZE;
        int fromX = (int) x / tile;
        int toX = (int) (x + PowerUp.SIZE - 1) / tile;
        int fromY = (int) y / tile;
        int toY = (int) (y + PowerUp.SIZE - 1) / tile;

        for (int tx = fromX; tx <= toX; tx++) {
            for (int ty = fromY; ty <= toY; ty++) {
                if (tx < 0 || tx >= map.length || ty < 0 || ty >= map[0].length) return true;
                if (map[tx][ty]) return true;
            }
        }
        return false;
    }

    private PowerUp create(int type, double x, double y) {
        return switch (type) {
            case 0 -> new Heal(x, y, 25);
            case 1 -> new Shield(x, y, 50);
            case 2 -> new DamageBoost(x, y, 10000);
            case 3 -> new Ammo(x, y);
            default -> new MaxAmmo(x, y);
        };
    }

package model;

import java.util.ArrayDeque;

/**
 * Calcula a quÃ© zonas del mapa puede llegar realmente el jugador desde donde empieza.
 * Usa la MISMA regla de colisiÃ³n con la que el jugador se mueve (Character.isCollidingWithMap),
 * asÃ­ que si acÃ¡ figura como alcanzable, el jugador puede llegar caminando.
 */
public class ReachableArea {

    private static final int STEP = 16;          // resoluciÃ³n de la grilla (px) // Thiago: cambio a 16 por motivos de rendimiento
    private static final int PICKUP_REACH = 32; // cuÃ¡nto puede estar el centro del jugador del centro del Ã­tem
    private static final int MIN_CELLS = 100;   // si da menos, el cÃ¡lculo no es confiable

    private final boolean[][] reachable; // [col][fila] -> posiciÃ³n (esq. sup. izq.) del jugador
    private final int cols;
    private final int rows;
    private final int reachableCells;

    public ReachableArea(GameState state) {
        Player player = state.getPlayer();
        cols = (state.getMapWidth() - player.getWidth()) / STEP + 1;
        rows = (state.getMapHeight() - player.getHeight()) / STEP + 1;
        reachable = new boolean[Math.max(cols, 0)][Math.max(rows, 0)];

        if (cols <= 0 || rows <= 0) {
            reachableCells = 0;
            return;
        }

        int startCol = clamp(Math.round((float) player.getX() / STEP), 0, cols - 1);
        int startRow = clamp(Math.round((float) player.getY() / STEP), 0, rows - 1);

        ArrayDeque<int[]> queue = new ArrayDeque<>();
        reachable[startCol][startRow] = true;
        queue.add(new int[]{startCol, startRow});

        int count = 1;
        int[][] directions = {{1, 0}, {-1, 0}, {0, 1}, {0, -1}};

        while (!queue.isEmpty()) {
            int[] cell = queue.poll();
            for (int[] d : directions) {
                int c = cell[0] + d[0];
                int r = cell[1] + d[1];
                if (c < 0 || r < 0 || c >= cols || r >= rows || reachable[c][r]) continue;
                if (player.isCollidingWithMap(c * STEP, r * STEP, state)) continue;

                reachable[c][r] = true;
                queue.add(new int[]{c, r});
                count++;
            }
        }
        reachableCells = count;
    }

    /** false si el cÃ¡lculo no sirviÃ³ (ej: el jugador empieza encerrado); en ese caso no hay que usarlo. */
    public boolean isUsable() {
        return reachableCells >= MIN_CELLS;
    }

    /** true si el jugador puede llegar caminando lo bastante cerca como para agarrar un Ã­tem centrado en (itemX, itemY). */
    public boolean canPickUpAt(double itemCenterX, double itemCenterY, Player player) {
        double playerLeft = itemCenterX - player.getWidth() / 2.0;
        double playerTop = itemCenterY - player.getHeight() / 2.0;

        int fromCol = (int) Math.floor((playerLeft - PICKUP_REACH) / STEP);
        int toCol = (int) Math.ceil((playerLeft + PICKUP_REACH) / STEP);
        int fromRow = (int) Math.floor((playerTop - PICKUP_REACH) / STEP);
        int toRow = (int) Math.ceil((playerTop + PICKUP_REACH) / STEP);

        for (int c = Math.max(fromCol, 0); c <= Math.min(toCol, cols - 1); c++) {
            for (int r = Math.max(fromRow, 0); r <= Math.min(toRow, rows - 1); r++) {
                if (reachable[c][r]) return true;
            }
        }
        return false;
    }

    private static int clamp(int value, int min, int max) {
        return Math.max(min, Math.min(value, max));
    }
}

            statement.execute("CREATE TABLE IF NOT EXISTS matches ("
                    + "id INTEGER PRIMARY KEY AUTOINCREMENT, "

                    + "zombies_killed INTEGER NOT NULL, "
                    + "damage_dealt INTEGER NOT NULL, "

        }

        String sql = "INSERT INTO matches (match_date, survival_time_ms, zombies_killed, damage_dealt, damage_taken, victory)"
                + "VALUES (?, ?, ?, ?, ?)";

        try (PreparedStatement statement = this.connection.prepareStatement(sql)) {
            statement.setString(1, match.getDate().toString());
            statement.setLong(2, match.getSurvivalTimeMs());
            statement.setInt(3, match.getZombiesKilled());
            statement.setInt(4, match.getDamageDealt());
            statement.setInt(5, match.getDamageTaken());
            statement.setInt(6, match.isVictory() ? 1 : 0);

        } catch (SQLException e) {
            return false;
        }
    }

        if (limit <= 0) {
            return result;
        }

                + "damage_taken, victory FROM matches WHERE victory = 1 "
                + " ORDER BY survival_time_ms ASC LIMIT ?";

        try (PreparedStatement statement = this.connection.prepareStatement(sql)) {

    public void drawMuzzleFlash(Graphics2D g2, double x, double y, int w, int h, double angle, String weaponType, long lastShoot, double muzzleForward, double muzzleSide) {

        boolean shotgun = weaponType.equals("Shotgun");
        long duration = shotgun ? SHOTGUN_FLASH_MS : FLASH_MS;
        long elapsed = System.currentTimeMillis() - lastShoot;
        if (elapsed < 0 || elapsed > duration) return;
 
        double life = 1.0 - (double) elapsed / duration; // 1 al disparar -> 0 al apagarse
 
        double centerX = x + w / 2.0;
        double centerY = y + h / 2.0;
 
        AffineTransform old = g2.getTransform();
        Paint oldPaint = g2.getPaint();
        Stroke oldStroke = g2.getStroke();
 
        // A partir de acÃ¡, el eje X apunta hacia donde mira el jugador y (0,0) es la boca del arma
        g2.translate(centerX, centerY);
        g2.rotate(angle);
        g2.translate(muzzleForward, muzzleSide);
 
        // Misma semilla durante todo el destello: la forma se mantiene y solo se desvanece
        Random rnd = new Random(lastShoot);
 
        drawFlashGlow(g2, life, shotgun);
        drawFlashSpikes(g2, rnd, life, shotgun);
        drawFlashSparks(g2, rnd, life, shotgun);
 
        g2.setTransform(old);
        g2.setPaint(oldPaint);
        g2.setStroke(oldStroke);
    }
 
    /** Resplandor redondo + luz tenue que "ilumina" la zona oscura alrededor. */
    private void drawFlashGlow(Graphics2D g2, double life, boolean shotgun) {
        Point2D center = new Point2D.Double(0, 0);
        float[] fractions = {0f, 0.35f, 1f};
 
        // IluminaciÃ³n ambiente
        double ambient = shotgun ? 190 : 140;
        g2.setPaint(new RadialGradientPaint(center, (float) ambient, fractions, new Color[]{
                new Color(255, 170, 70, (int) (70 * life)),
                new Color(255, 120, 30, (int) (35 * life)),
                new Color(255, 100, 0, 0)}));
        g2.fill(new Ellipse2D.Double(-ambient, -ambient, ambient * 2, ambient * 2));
 
        // NÃºcleo brillante
        double radius = (shotgun ? 70 : 50) * (0.6 + 0.4 * life);
        g2.setPaint(new RadialGradientPaint(center, (float) radius, fractions, new Color[]{
                new Color(255, 245, 190, (int) (230 * life)),
                new Color(255, 150, 40, (int) (150 * life)),
                new Color(255, 100, 0, 0)}));
        g2.fill(new Ellipse2D.Double(-radius, -radius, radius * 2, radius * 2));
    }
 
    /** Picos de fuego en abanico hacia adelante. */
    private void drawFlashSpikes(Graphics2D g2, Random rnd, double life, boolean shotgun) {
        int count = shotgun ? 9 : 6;
        double spread = shotgun ? 0.75 : 0.5;     // apertura del abanico (rad)
        double baseLength = shotgun ? 70 : 48;
 
        Path2D outer = new Path2D.Double();
        Path2D inner = new Path2D.Double();
 
        for (int i = 0; i < count; i++) {
            // El primer pico va derecho (el mÃ¡s largo); el resto se reparte en abanico
            double angle = (i == 0) ? 0 : (rnd.nextDouble() * 2 - 1) * spread;
            double length = baseLength * (i == 0 ? 1.2 : 0.45 + rnd.nextDouble() * 0.6) * (0.5 + 0.5 * life);
            double halfWidth = (i == 0 ? 7 : 4) * (0.6 + 0.4 * life);
 
            addSpike(outer, angle, length, halfWidth);
            addSpike(inner, angle, length * 0.55, halfWidth * 0.5);
        }
 
        g2.setPaint(new Color(255, 160, 40, (int) (230 * life)));
        g2.fill(outer);
        g2.setPaint(new Color(255, 250, 215, (int) (255 * life)));
        g2.fill(inner);
    }
 
    /** TriÃ¡ngulo fino desde la boca del arma hacia el Ã¡ngulo indicado. */
    private void addSpike(Path2D path, double angle, double length, double halfWidth) {
        double dx = Math.cos(angle);
        double dy = Math.sin(angle);
        double px = -dy * halfWidth; // perpendicular
        double py = dx * halfWidth;
 
        path.moveTo(px, py);
        path.lineTo(dx * length, dy * length);
        path.lineTo(-px, -py);
        path.closePath();
    }
 
    /** Chispas chicas que salen disparadas y se alejan mientras el destello se apaga. */
    private void drawFlashSparks(Graphics2D g2, Random rnd, double life, boolean shotgun) {
        int count = shotgun ? 9 : 5;
        double progress = 1.0 - life; // 0 -> 1
 
        g2.setStroke(new BasicStroke(1.6f, BasicStroke.CAP_ROUND, BasicStroke.JOIN_ROUND));
 
        for (int i = 0; i < count; i++) {
            double angle = (rnd.nextDouble() * 2 - 1) * 0.9;
            double maxDist = 30 + rnd.nextDouble() * (shotgun ? 70 : 45);
            double start = 6 + progress * maxDist;
            double end = start + 5 + rnd.nextDouble() * 7;
 
            g2.setPaint(new Color(255, 215, 120, (int) (255 * life)));
            g2.draw(new Line2D.Double(
                    Math.cos(angle) * start, Math.sin(angle) * start,
                    Math.cos(angle) * end, Math.sin(angle) * end));
        }
    }
 
    private static Color alpha(Color c, int a) {
        return new Color(c.getRed(), c.getGreen(), c.getBlue(), Math.max(0, Math.min(255, a)));
    }

package view.hud;

import java.awt.Color;
import java.awt.Font;
import java.awt.Graphics2D;

/**
 * Clase base de todo lo que se dibuja en el HUD.
 * El Hud los trata a todos igual (update + draw); cada hijo decide cÃ³mo se ve y cÃ³mo reacciona.
 */
public abstract class HudElement {

    protected static final int MARGIN = 24;
    protected static final Font FONT = new Font("SansSerif", Font.BOLD, 14);

    /** Se llama una vez por frame con los datos actuales y el tiempo transcurrido en segundos. */
    public void update(HudData data, double dt) {
    }

    /** Se llama cuando el jugador agarra un power up, para que el elemento reaccione. */
    public void onPickup(PowerUpStyle style) {
    }

    public abstract void draw(Graphics2D g, int screenW, int screenH);

    // ---------- utilidades compartidas por los hijos ----------

    protected static double clamp01(double v) {
        return Math.max(0, Math.min(1, v));
    }

    protected static Color alpha(Color c, double a) {
        return new Color(c.getRed(), c.getGreen(), c.getBlue(), (int) (clamp01(a) * 255));
    }

    protected static Color mix(Color a, Color b, double t) {
        t = clamp01(t);
        return new Color(
                (int) (a.getRed() + (b.getRed() - a.getRed()) * t),
                (int) (a.getGreen() + (b.getGreen() - a.getGreen()) * t),
                (int) (a.getBlue() + (b.getBlue() - a.getBlue()) * t));
    }

    protected static double easeOutBack(double t) {
        double c1 = 1.70158, c3 = c1 + 1;
        return 1 + c3 * Math.pow(t - 1, 3) + c1 * Math.pow(t - 1, 2);
    }

    protected static void shadowText(Graphics2D g, String s, float x, float y, Color c) {
        g.setColor(new Color(0, 0, 0, 190));
        g.drawString(s, x + 1.5f, y + 1.5f);
        g.setColor(c);
        g.drawString(s, x, y);
    }

    protected static int textWidth(Graphics2D g, String s) {
        return g.getFontMetrics().stringWidth(s);
    }
}

/** ViÃ±eta de color en los bordes de la pantalla: rojo al recibir daÃ±o, celeste si lo absorbe el escudo, color del power up al agarrarlo. */
public class ScreenFlash extends HudElement {

    private static class Pulse {
        final Color color;
        double strength;
        final double decay;

        Pulse(Color color, double strength, double decay) {
            this.color = color;
            this.strength = strength;
            this.decay = decay;
        }
    }

    private final List<Pulse> pulses = new ArrayList<>();
    private int lastHealth = -1;
    private int lastShield = -1;
    private double lowHealth; // 0 sano .. 1 casi muerto
    private double time;

    @Override
    public void update(HudData data, double dt) {
        time += dt;
        if (lastHealth >= 0) {
            if (data.health() < lastHealth) {
                pulses.add(new Pulse(new Color(220, 20, 30), 0.95, 1.5));
            } else if (data.shield() < lastShield) {
                pulses.add(new Pulse(PowerUpStyle.SHIELD.color(), 0.6, 2.0));
            }
        }
        lastHealth = data.health();
        lastShield = data.shield();

        for (Pulse p : pulses) p.strength -= p.decay * dt;
        pulses.removeIf(p -> p.strength <= 0);

        double ratio = data.maxHealth() > 0 ? data.health() / (double) data.maxHealth() : 1;
        lowHealth = data.health() > 0 ? clamp01((0.3 - ratio) / 0.3) : 0;
    }

    @Override
    public void onPickup(PowerUpStyle style) {
        pulses.add(new Pulse(style.color(), 0.5, 2.2));
    }

    @Override
    public void draw(Graphics2D g, int screenW, int screenH) {
        for (Pulse p : pulses) {
            vignette(g, screenW, screenH, p.color, p.strength * 0.8);
        }
        if (lowHealth > 0) {
            double beat = 0.55 + 0.45 * Math.sin(time * 5);
            vignette(g, screenW, screenH, new Color(200, 0, 15), lowHealth * beat * 0.55);
        }
    }

    private void vignette(Graphics2D g, int w, int h, Color c, double strength) {
        float radius = (float) Math.hypot(w, h) / 2f;
        g.setPaint(new RadialGradientPaint(
                new Point2D.Float(w / 2f, h / 2f), radius,
                new float[]{0.4f, 1f},
                new Color[]{alpha(c, 0), alpha(c, strength)}));
        g.fillRect(0, 0, w, h);
    }
}

public class ShieldBar extends StatBar {

    private static final int BAR_W = 300;
    private static final int BAR_H = 18;
    private static final Color LEFT = new Color(35, 140, 225);
    private static final Color RIGHT = new Color(130, 240, 255);

    private double appear; // 0 oculto .. 1 visible

    @Override
    public void update(HudData data, double dt) {
        track(data.shield(), data.maxShield(), dt);
        double want = (target > 0 || shown > 0.5) ? 1 : 0;
        appear += (want - appear) * Math.min(1, dt * 8);
    }

    @Override
    protected void onLoss(double amount) {
        triggerFlash(Color.WHITE);
        shake = 1;
    }

    @Override
    public void onPickup(PowerUpStyle style) {
        if (style == PowerUpStyle.SHIELD) {
            triggerFlash(style.color());
            pop = 1;
        }
    }

    @Override
    public void draw(Graphics2D g, int screenW, int screenH) {
        if (appear < 0.02) return;

        g.setComposite(AlphaComposite.SrcOver.derive((float) clamp01(appear)));
        int x = MARGIN + shakeOffset();

package view.hud;

import java.awt.BasicStroke;
import java.awt.Color;
import java.awt.GradientPaint;
import java.awt.Graphics2D;
import java.awt.Shape;
import java.awt.geom.RoundRectangle2D;

/**
 * Barra animada genÃ©rica (la usan la vida y el escudo).
 * Se encarga de lo comÃºn: valor suavizado, estela blanca del daÃ±o recibido, flash y temblor.
 * Los hijos solo definen colores, icono y quÃ© hacer al perder/ganar.
 */
abstract class StatBar extends HudElement {

    protected double shown;   // valor que se ve (persigue al real con suavizado)
    protected double ghost;   // estela: se queda en el valor viejo y baja despuÃ©s de una pausa
    protected double target;  // valor real
    protected double max = 1;
    protected double flash;   // 0..1, destello sobre la barra
    protected Color flashColor = Color.WHITE;
    protected double shake;   // 0..1, temblor
    protected double pop;     // 0..1, "salto" del icono
    protected double time;

    private double ghostDelay;
    private boolean started;

    protected void track(int value, int maxValue, double dt) {
        time += dt;
        max = Math.max(1, maxValue);
        if (!started) {
            shown = ghost = target = value;
            started = true;
        }
        if (value < target) {
            ghostDelay = 0.5;
            onLoss(target - value);
        } else if (value > target) {
            onGain(value - target);
        }
        target = value;

        shown += (target - shown) * Math.min(1, dt * 10);
        if (Math.abs(target - shown) < 0.05) shown = target;

        if (shown >= ghost) {
            ghost = shown;
        } else if (ghostDelay > 0) {
            ghostDelay -= dt;
        } else {
            ghost += (shown - ghost) * Math.min(1, dt * 4);
        }

        flash = Math.max(0, flash - dt * 2.2);
        shake = Math.max(0, shake - dt * 3);
        pop = Math.max(0, pop - dt * 2.5);
    }

    protected void onLoss(double amount) {
    }

    protected void onGain(double amount) {
    }

    protected void triggerFlash(Color color) {
        flashColor = color;
        flash = 1;
    }

    protected int shakeOffset() {
        return shake > 0 ? (int) (Math.sin(time * 70) * 5 * shake) : 0;
    }

    protected void drawBar(Graphics2D g, int x, int y, int w, int h, Color left, Color right, int segments) {
        RoundRectangle2D shape = new RoundRectangle2D.Double(x, y, w, h, h, h);
        g.setColor(new Color(0, 0, 0, 160));
        g.fill(shape);

        Shape oldClip = g.getClip();
        g.clip(shape);

        // estela del daÃ±o
        g.setColor(new Color(255, 255, 255, 150));
        g.fillRect(x, y, (int) (w * clamp01(ghost / max)), h);

        // relleno con degradÃ©, brillo superior y punta luminosa
        int fillW = (int) (w * clamp01(shown / max));
        if (fillW > 0) {
            g.setPaint(new GradientPaint(x, y, left, x + w, y, right));
            g.fillRect(x, y, fillW, h);
            g.setPaint(new GradientPaint(0, y, new Color(255, 255, 255, 95), 0, y + h / 2f, new Color(255, 255, 255, 0)));
            g.fillRect(x, y, fillW, h / 2);
            g.setColor(new Color(255, 255, 255, 170));
            g.fillRect(x + fillW - 2, y, 2, h);
        }

        // divisiones
        if (segments > 1) {
            g.setColor(new Color(0, 0, 0, 75));
            for (int i = 1; i < segments; i++) {
                g.fillRect(x + w * i / segments - 1, y, 2, h);
            }
        }

        if (flash > 0) {
            g.setColor(alpha(flashColor, flash * 0.65));
            g.fill(shape);
        }
        g.setClip(oldClip);

        g.setStroke(new BasicStroke(1.5f));
        g.setColor(new Color(255, 255, 255, 80));
        g.draw(shape);
    }
}
