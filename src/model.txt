import java.awt.Rectangle;

public class Bullet extends Entity {
    private final double dirX;
    private final double dirY;
    private final double speed;
    private final double damage;
    private final BulletKind kind;
    private boolean active; // booleano para saber si la bala debe 'desaparecer'
    private int lifeSpanTicks;
    private int age;        // ticks que lleva viva (la vista lo usa para que la estela "salga" del arma)

    public Bullet(double x, double y, int width, int height, double dirX, double dirY, double damage, int lifeSpanTicks) {
        this(x, y, width, height, dirX, dirY, damage, lifeSpanTicks, BulletKind.STANDARD);
    }

    public Bullet(double x, double y, int width, int height, double dirX, double dirY, double damage, int lifeSpanTicks, BulletKind kind) {
        super(x, y, width, height);
        this.dirX = dirX;
        this.dirY = dirY;
        this.speed = 12.0;
        this.damage = damage;
        this.active = true;
        this.lifeSpanTicks = lifeSpanTicks;
        this.kind = kind;
        this.age = 0;
    }

    public boolean isActive() {
        return active;
    }
    public void setActive(boolean active) {
        this.active = active;
    }
    public double getDamage(){
        return damage;
    }
    public double getDirX() {
        return dirX;
    }
    public double getDirY() {
        return dirY;
    }
    public BulletKind getKind() {
        return kind;
    }
    public int getAge() {
        return age;
    }
    public double getSpeed() {
        return speed;
    }

    public void update(int mapWidth, int mapHeight) {
        this.x += dirX * speed; 
        this.y += dirY * speed;

        this.age++;
        this.lifeSpanTicks--;

        if (this.lifeSpanTicks <= 0) {
            this.active = false;
        }

        // Desactivar la bala si sale de los límites de la ventana
        if (x < 0 || x > mapWidth || y < 0 || y > mapHeight) {
            active = false;
        }
    }

    public Rectangle getBounds(){
        return new Rectangle((int) x, (int) y, (int) width, (int) height);
    }
}

package model;

public abstract class Character extends Entity {

    protected double speed;
    private int health;
    private int maxHealth;

    public Character(double x, double y, int width, int height, double speed, int health, int maxHealth) {
        super(x, y, width, height);
        this.speed = speed;
        this.maxHealth = maxHealth;
        this.health = health;
    }

    public int getHealth() {
        return health;
    }
    public void setHealth(int health) {
        this.health = health;
    }

    public int getMaxHealth() {
        return maxHealth;
    }

    protected void move(double dx, double dy, GameState state) {
        if (dx != 0 || dy != 0) {
            double length = Math.sqrt(dx * dx + dy * dy);
            double moveX = (dx / length) * speed;
            double moveY = (dy / length) * speed;

            if (!isCollidingWithMap(x + moveX, y, state)) x += moveX;
            if (!isCollidingWithMap(x, y + moveY, state)) y += moveY;
        }
    }

    protected boolean isCollidingWithMap(double nextX, double nextY, GameState state) {
        boolean[][] map = state.getCollisionMatrix();
        if (map == null) return false;

        int tile_size = GameState.TILE_SIZE;

        // Hitbox lógica (más chica que sprite visual)
        int leftTile = (int) (nextX + 15) / tile_size;
        int rightTile = (int) (nextX + width - 15) / tile_size;
        int topTile = (int) (nextY + 30) / tile_size;
        int bottomTile = (int) (nextY + height - 5) / tile_size;

        if (leftTile < 0 || rightTile >= map.length || topTile < 0 || bottomTile >= map[0].length) {
            return true;
        }

        return map[leftTile][topTile] || map[rightTile][topTile] ||
           map[leftTile][bottomTile] || map[rightTile][bottomTile];
    }

    protected boolean canMove(double dx, double dy, GameState state) {
        if (dx == 0 && dy == 0) return false;

        double length = Math.hypot(dx, dy);
        double moveX = (dx / length) * speed;
        double moveY = (dy / length) * speed;

        return !isCollidingWithMap(x + moveX, y + moveY, state);
    }

    public double getSpeed() {
        return speed;
    }

    public void setSpeed(double speed) {
        this.speed = speed;
    }

    public void takeDamage(int amount) {
        if (amount < this.health) {
            health -= amount;
        } else {
            this.health = 0;
        }
    }

    public boolean isAlive() {
        return health > 0;
    }
}

===== CollisionSystem.java =====
package model;

public class CollisionSystem {

    public void update (GameState state) {
        checkBulletZombieCollision(state);
        chekBulletMapCollision(state);
        checkPoisonPlayerCollision(state);
        checkZombiePlayerCollision(state);
    }

    private void checkBulletZombieCollision (GameState state) {
        for (Bullet bullet : state.getPlayerBullets()) {
            if (!bullet.isActive()) continue;

            for (Enemy zombie : state.getEnemies()) {
                if (zombie.isAlive() && bullet.collidesWith(zombie)) {
                    bullet.setActive(false);
                    zombie.takeDamage((int) bullet.getDamage());
                    break;
                }
            }
        }
    }

    private void chekBulletMapCollision (GameState state) {
        boolean[][] map = state.getCollisionMatrix();
        if (map == null) return;

        int tileSize = GameState.TILE_SIZE;

        for (Bullet bullet : state.getPlayerBullets()) {
            if (!bullet.isActive()) continue;

            int tileX = (int) (bullet.getX() + bullet.getWidth() / 2.0) / tileSize;
            int tileY = (int) (bullet.getY() + bullet.getHeight() / 2.0) / tileSize;

            if (tileX >= 0 && tileX < map.length && tileY >= 0 && tileY < map[0].length) {
                if (map[tileX][tileY]) {
                    bullet.setActive(false);
                }
            }
        }

        for (PoisonBullet poison : state.getPoisonBullets()) {
            if (!poison.isActive()) continue;

            int tileX = (int) (poison.getX() + poison.getWidth() / 2.0) / tileSize;
            int tileY = (int) (poison.getY() + poison.getHeight() / 2.0) / tileSize;

            if (tileX >= 0 && tileX < map.length && tileY >= 0 && tileY < map[0].length) {
                if (map[tileX][tileY]) {
                    poison.setActive(false);
                }
            }
        }
    }

    private void checkPoisonPlayerCollision (GameState state) {
        Player player = state.getPlayer();
        if (!player.isAlive()) return;

        for (PoisonBullet poison : state.getPoisonBullets()) {
            if (poison.isActive() && poison.collidesWith(player)) {
                poison.setActive(false);
                poison.applyTo(player);//POLIMORFISMO: cada bala sabe qué hacer
            }
        }
    }

    private void checkZombiePlayerCollision(GameState state) {
        Player player = state.getPlayer();
        if (!player.isAlive()) return;

        for (Enemy zombie : state.getEnemies()) {
            if (!zombie.isAlive()) continue;

            if (zombie.collidesWith(player)) {

                if (zombie.canAttack()) {
                    player.takeDamage(zombie.attack());
                    zombie.resetAttackCooldown();
                }
            }
        }
    }
}

    public void avoidOverlap(GameState state) {

            if (this == otherZombie || !otherZombie.isAlive()) continue;

            if (this.collidesWith(otherZombie)) {

                double dx = this.getX() - otherZombie.getX();
                double dy = this.getY() - otherZombie.getY();
                double distance = Math.sqrt(dx * dx + dy * dy);

                    }

===== EnemySpawner.java =====
package model;

import java.util.Random;

public class EnemySpawner {
    private long lastSpawnTime;
    private int spawnInterval;
    private int zombieSpawned;
    private final int waveLimit;

    public EnemySpawner(int spawnInterval, int waveLimit){
        this.spawnInterval = spawnInterval;
        this.waveLimit = waveLimit;
        this.lastSpawnTime = System.currentTimeMillis();
        this.zombieSpawned = 0;
    }
    
    public void update(GameState state){
        long currentTime = System.currentTimeMillis();

        if (currentTime - lastSpawnTime >= spawnInterval && zombieSpawned < waveLimit) {
            Random random = new Random();
            int tipoZombie = random.nextInt(4);
            int mapWidth = state.getMapWidth();
            int mapHeight = state.getMapHeight();

        
            if (mapWidth <= 100 || mapHeight <= 100) return;

            int spawnX = random.nextInt(mapWidth - 100);
            int spawnY = random.nextInt(mapHeight - 100);

            Enemy newZombie;

            if (tipoZombie == 0) {
                newZombie = new TankZombie(spawnX, spawnY);
            } else if (tipoZombie == 1) {
                newZombie = new FastZombie(spawnX, spawnY);
            } else if (tipoZombie == 2) {
                newZombie = new PoisonZombie(spawnX, spawnY, state.getPoisonBullets());
            } else {
                newZombie = new NormalZombie(spawnX, spawnY);
            }

            state.addEnemy(newZombie);
            lastSpawnTime = currentTime;
            zombieSpawned++;
        }
    }
    public void reset() {
    this.zombieSpawned = 0;
    this.lastSpawnTime = System.currentTimeMillis();
}

    public boolean isWaveComplete(){
        return zombieSpawned >= waveLimit;
    }
}

package model;

import java.awt.*;

public abstract class Entity {

    protected double x;
    protected double y;
    protected int width;
    protected int height;

    public Entity(double x, double y, int width, int height) {
        this.x = x;
        this.y = y;
        this.width = width;
        this.height = height;
    }

    public double getX() {
        return x;
    }

    public double getY() {
        return y;
    }

    public int getWidth() {
        return width;
    }

    public int getHeight() {
        return height;
    }

    public Rectangle getBounds() {
        return new Rectangle((int) x, (int) y, width, height);
    }

    public boolean collidesWith(Entity e) {
        if (e == null) return false;

        return this.x < e.getX() + e.getWidth() &&
           this.x + this.width > e.getX() &&
           this.y < e.getY() + e.getHeight() &&
           this.y + this.height > e.getY();
    }
}

package model;

public abstract class FireArm extends Weapon implements AmmoHolder {

    protected int actualMunition;
    private final int maxMunition;

    public FireArm(int damage, long cooldown, int actualMunition, int maxMunition){
        super(damage);
        this.cooldown = cooldown;
        this.actualMunition = actualMunition;
        this.maxMunition = maxMunition;
    }

    @Override
    public boolean attack(Player player, GameState state){
        long currentTime = System.currentTimeMillis();

        if (this.actualMunition > 0 && currentTime - this.lastShoot >= this.cooldown) {

            this.actualMunition--;
            this.lastShoot = currentTime;

            double dirX = Math.cos(player.getAngle());
            double dirY = Math.sin(player.getAngle());

            double muzzleX = getMuzzleX(player);
            double muzzleY = getMuzzleY(player);

            double totalDamage = this.getDamage() + player.getDamageBonus();
            Bullet bullet = new Bullet(muzzleX, muzzleY, 6, 6, dirX, dirY, totalDamage, 60);

            state.addPlayerBullet(bullet);

            return true;
        }
        return false;
    }

    @Override
    public void addAmmo(int amount) {
        // El math.min asegura que no se pase de la munición máxima
        this.actualMunition = Math.min(this.actualMunition + amount, this.maxMunition);
    }

    @Override
    public void refillAmmo() {
        this.actualMunition = this.maxMunition;
    }

    public int getActualMunition() {
        return actualMunition;
    }
    public int getMaxMunition() {
        return maxMunition;
    }
}

===== GameModel.java =====
package model;

import java.util.List;

public class GameModel {

    private final GameState state;
    private final CollisionSystem collisionSystem;
    private final EnemySpawner spawner;
    private final PowerUpSpawner powerUpSpawner;

    public enum gameStatus {PLAYING, PAUSED, BOSS_INTRO, LEVEL_COMPLETED, GAME_OVER, GAME_WON}
    private gameStatus status = gameStatus.PLAYING;
    private int actualLevel = 1;
    private final int totalLevels = 2;
    private boolean bossSpawned = false;

    // Intro del jefe: 2 mensajes de INTRO_STEP_MS cada uno
    private static final int INTRO_STEPS = 2;
    private static final long INTRO_STEP_MS = 3500;
    private long bossIntroStart;

    public GameModel(GameState state) {
        this.state = state;
        this.collisionSystem = new CollisionSystem();

        // Inicializamos los spawners
        this.spawner = new EnemySpawner(2000, 10);
        this.powerUpSpawner = new PowerUpSpawner(state.getPowerUps(), 8000, 3);
       // state.addEnemy(new FinalBoss(1000, 300, state.getPoisonBullets())); //PRUEBA PARA JEFE ZOMBIE FINAL
    }

    public void update(int screenWidth, int screenHeight) {
        if (status == gameStatus.BOSS_INTRO) {
            updateBossIntro();
            return;
        }
        if (status != gameStatus.PLAYING) return;

        // Mover jugador
        Player player = state.getPlayer();
        if (!player.isAlive()) {
            status = gameStatus.GAME_OVER;
            return;
        }

        player.update(state);

        int mapWidth = state.getMapWidth();
        int mapHeight = state.getMapHeight();

        // Mover balas
        for (Bullet b : state.getPlayerBullets()) b.update(mapWidth, mapHeight);
        for (Bullet p: state.getPoisonBullets()) p.update(mapWidth, mapHeight);

        // Spawners
        spawner.update(state);
        powerUpSpawner.update(screenWidth, screenHeight);

        // Mover zombies
        for (Enemy zombie : state.getEnemies()) {
            if (zombie.isAlive()) {
                zombie.update(player, state);
            }
        }

        // Recolección de power-ups
        for (PowerUp p : state.getPowerUps()) {
            if (p.isActive() && player.collidesWith(p)) {
                p.collect(player);
            }
        }

        // Sistema de colisiones (daño)
        collisionSystem.update(state);

        // Limpia los inactivos
        state.purgeInactive();

        // Oleada terminada -> intro del jefe; jefe derrotado -> fin de nivel
        if (state.getEnemies().isEmpty() && spawner.isWaveComplete()) {
            if (!bossSpawned) {
                startBossIntro();
            } else if (actualLevel < totalLevels) {
                status = gameStatus.LEVEL_COMPLETED;
            } else {
                status = gameStatus.GAME_WON;
            }
        }
    }

    private void startBossIntro() {
        status = gameStatus.BOSS_INTRO;
        bossIntroStart = System.currentTimeMillis();
    }

    private void updateBossIntro() {
        long elapsed = System.currentTimeMillis() - bossIntroStart;
        if (elapsed >= INTRO_STEP_MS * INTRO_STEPS) {
            spawnBoss();
            status = gameStatus.PLAYING;
        }
    }

    private void spawnBoss() {
        state.addEnemy(new FinalBoss(1000, 300, state.getPoisonBullets()));
        bossSpawned = true;
    }

    /** Qué mensaje toca mostrar: 0 = felicitación, 1 = "Preparate para el Jefe". */
    public int getBossIntroStep() {
        long elapsed = System.currentTimeMillis() - bossIntroStart;
        return (int) Math.min(INTRO_STEPS - 1, elapsed / INTRO_STEP_MS);
    }

    /** Avance (0.0 a 1.0) dentro del mensaje actual, para el efecto de aparecer y desvanecer. */
    public double getBossIntroProgress() {
        long elapsed = System.currentTimeMillis() - bossIntroStart;
        return Math.min(1.0, (elapsed % INTRO_STEP_MS) / (double) INTRO_STEP_MS);
    }

    public void togglePause() {
        if (status == gameStatus.PLAYING) {
            status = gameStatus.PAUSED;
        } else if (status == gameStatus.PAUSED) {
            status = gameStatus.PLAYING;
        }
    }

    public void playerInteract() {
        if (status != gameStatus.PLAYING) return;

        List<WeaponItem> groundWeapons = state.getGroundWeapons();
        Player player = state.getPlayer();

        for (int i = 0; i < groundWeapons.size(); i++) {
            WeaponItem item = groundWeapons.get(i);

            if (player.collidesWith(item)) {
                FireArm oldWeapon = player.getPrimaryWeapon();
                player.equipWeapon(item.getWeapon());
                groundWeapons.remove(i);

                if (oldWeapon != null) {
                    WeaponItem droppedWeapon = new WeaponItem(player.getX(), player.getY(), oldWeapon);
                    groundWeapons.add(droppedWeapon);
                }
                break;
            }
        }
    }

    public void nextLevel() {
        if (status != gameStatus.LEVEL_COMPLETED) return;
        actualLevel++;
        bossSpawned = false;
        spawner.reset();
        state.purgeInactive();
        status = gameStatus.PLAYING;
    }

    public GameState getState() {
        return state;
    }
    public gameStatus getStatus() {
        return status;
    }
}

package model;

import java.io.InputStream;
import java.util.Scanner;

public class MapLoader {

    public static boolean[][] loadCollisionMatrix(String filePath, int cols, int rows) {
        boolean[][] matrix = new boolean[cols][rows];
        try (InputStream is = MapLoader.class.getResourceAsStream(filePath)) {
            if (is == null) {
                System.err.println("No se encontró el mapa en: " + filePath);
                return matrix;
            }
            Scanner scanner = new Scanner(is, "UTF-8").useDelimiter("\\A");
            String json = scanner.hasNext() ? scanner.next() : "";

            // Buscamos el array en el JSON (tmj)
            int dataIndex = json.indexOf("\"data\":[");
            if (dataIndex != -1) {
                int startBracket = json.indexOf("[", dataIndex);
                int endBracket = json.indexOf("]", startBracket);
                String dataString = json.substring(startBracket + 1, endBracket);
                String[] values = dataString.split(",");

                int col = 0;
                int row = 0;
                for (String val: values) {
                    long id = Long.parseLong(val.trim());
                    matrix[col][row] = (id > 0); // ">" true si es cuadro rojo
                    col++;
                    if (col >= cols) {
                        col = 0;
                        row++;
                    }
                }
            }
            scanner.close();
        } catch (Exception e) {
            System.err.println("Error leyendo el mapa: " + e.getMessage());
        }
        return matrix;
    }
}

===== MeleeWeapon.java =====
package model;

public class MeleeWeapon extends Weapon {

    private final double range = 100.0; // Píxeles de alcance del cuchillo
    private final double arcAngle = Math.toRadians(90); // 90 grados de arco de ataque

    public MeleeWeapon(int damage, long cooldown) {
        super(damage);
        this.cooldown = cooldown;
    }

    @Override
    public boolean attack(Player player, GameState state) {
        long currentTime = System.currentTimeMillis();

        if (currentTime - this.lastShoot >= this.cooldown) {
            this.lastShoot = currentTime;

            double totalDamage = this.getDamage() + player.getDamageBonus();
            boolean hitSomething = false;

            for (Enemy zombie : state.getEnemies()) {
                if (!zombie.isAlive()) continue;

                double playerCenterX = player.getX() + (player.getWidth() / 2.0);
                double playerCenterY = player.getY() + (player.getHeight() / 2.0);

                double zombieCenterX = zombie.getX() + (zombie.getWidth() / 2.0);
                double zombieCenterY = zombie.getY() + (zombie.getHeight() / 2.0);

                double dx = zombieCenterX - playerCenterX;
                double dy = zombieCenterY - playerCenterY;

                double distance = Math.hypot(dx, dy);

                if (distance <= range) {
                    double angleToZombie = Math.atan2(dy, dx); // Calculo el ángulo hacia el zombie

                    double angleDiff = angleToZombie - player.getAngle();
                    angleDiff = Math.atan2(Math.sin(angleDiff), Math.cos(angleDiff));

                    if (Math.abs(angleDiff) <= this.arcAngle / 2.0) {
                        zombie.takeDamage((int) totalDamage);
                        hitSomething = true;
                    }
                }
            }
            return hitSomething;
        }
        return false;
    }

    @Override
    public boolean isAutomatic() {
        return false;
    }
}

public class Player extends Character {

    private boolean moveUp;
    private boolean moveDown;

    private boolean isAttacking;

    // [POWERUP] estado que modifican los power ups
    private final List<Effect> effects = new ArrayList<>();
    private double damageBonus = 0; // daño extra que se suma a cada bala
    private int shield = 0;
    private static final int MAX_SHIELD = 100;
    private AmmoHolder ammoHolder; // arma actual (puede ser null hasta que exista Weapon)

    private double angle; // Ángulo en radianes hacia el cursor

    private final MeleeWeapon knife;
    private FireArm primaryWeapon; // Arma de fuego equipada
    private Weapon activeWeapon; // Puntero al arma que tiene en la mano actualmente

    public Player(double x, double y) {
        super(x, y, 64, 64, 5.5 , 100, 100);

        this.knife = new MeleeWeapon(50, 1000);
        this.activeWeapon = this.knife;
    }

    public void setMoveUp(boolean moveUp) {
        this.moveUp = moveUp;
    }

    public void setMoveDown(boolean moveDown) {
        this.moveDown = moveDown;
    }

    public void setMoveLeft(boolean moveLeft) {
        this.moveLeft = moveLeft;
    }

    public void setMoveRight(boolean moveRight) {
        this.moveRight = moveRight;
    }

    public void setAttacking(boolean attacking) {
        this.isAttacking = attacking;
    }

    public boolean isMoving() {
        return moveUp || moveDown || moveRight || moveLeft;
    }

    public double getAngle() {
        return angle;
    }

    public FireArm getPrimaryWeapon() {
        return primaryWeapon;
    }

    public Weapon getActiveWeapon() {
        return activeWeapon;
    }

    public void update(GameState state) {
        updateEffects(); // [POWERUP] vence los efectos temporales

        if (this.isAttacking && this.activeWeapon != null && this.activeWeapon.isAutomatic()) {
            this.attack(state);
        }

        double dx = 0;
        double dy = 0;
        if (moveUp) dy -= 1;
        if (moveDown) dy += 1;
        if (moveLeft) dx -= 1;
        if (moveRight) dx += 1;

        move(dx, dy, state);

        // Bordes del mapa calculados 
        if (x < 0) x = 0;
        if (x + width > state.getMapWidth()) x = state.getMapWidth() - width;
        if (y < 0) y = 0;
        if (y + height > state.getMapHeight()) y = state.getMapHeight() - height;
    }

    public void aimAt(double mouseX, double mouseY) {
        //Centro real del personaje
        double centerX = this.getX() + (this.getWidth() / 2.0);
        double centerY = this.getY() + (this.getHeight() / 2.0);

        double deltaX = mouseX - centerX;
        double deltaY = mouseY - centerY;

        this.angle = Math.atan2(deltaY, deltaX);
    }

    public void swapWeapon() {
        if (this.activeWeapon == this.knife && this.primaryWeapon != null) {
            this.activeWeapon = this.primaryWeapon;
            this.setAmmoHolder(this.primaryWeapon);
        } else {
            this.activeWeapon = this.knife;
            this.setAmmoHolder(null);
        }
    }

    public void equipWeapon(FireArm weapon) {
        this.primaryWeapon = weapon;
        this.activeWeapon = this.primaryWeapon;
        this.setAmmoHolder(weapon);
    }

    public void attack(GameState state) {
        if (this.activeWeapon != null) {
            this.activeWeapon.attack(this, state);
        }
    }

    public void addEffect(Effect effect) {
        effect.apply(this);
        effects.add(effect);
    }

    private void updateEffects() {
        Iterator<Effect> it = effects.iterator();
        while (it.hasNext()) {
            Effect e = it.next();
            e.update(this);
            if (e.isExpired()) {
                e.remove(this);
                it.remove();
            }
        }
    }

    public double getDamageBonus() {
        return damageBonus;
    }

    public void setDamageBonus(double damageBonus) {
        this.damageBonus = damageBonus;
    }

    public void heal(int amount) {
        setHealth(Math.min(getHealth() + amount, getMaxHealth()));
    }

    public void addShield(int points) {
        shield = Math.min(shield + points, MAX_SHIELD);
    }

    public int getShield() {
        return shield;
    }

    @Override
    public void takeDamage(int amount) {
        // el escudo absorbe primero
        int absorbed = Math.min(shield, amount);
        shield -= absorbed;
        super.takeDamage(amount - absorbed);
    }

    public void setAmmoHolder(AmmoHolder ammoHolder) {
        this.ammoHolder = ammoHolder;
    }

    public void addAmmo(int amount) {
        if (ammoHolder != null) ammoHolder.addAmmo(amount);
    }

    public void refillAmmo() {
        if (this.ammoHolder != null) {
            this.ammoHolder.refillAmmo();
        }
    }
}

package model;

import java.util.List;
import java.util.Random;

/** Genera power ups en posiciones aleatorias del mapa. Mismo estilo que EnemySpawner. */
public class PowerUpSpawner {

    private final List<PowerUp> powerUps;
    private final long spawnInterval; // ms entre cada spawn
    private final int maxOnMap;       // tope de power ups simultáneos
    private long lastSpawnTime;
    private final Random random = new Random();

    public PowerUpSpawner(List<PowerUp> powerUps, long spawnInterval, int maxOnMap) {
        this.powerUps = powerUps;
        this.spawnInterval = spawnInterval;
        this.maxOnMap = maxOnMap;
        this.lastSpawnTime = System.currentTimeMillis();
    }

    public void update(int mapWidth, int mapHeight) {
        long now = System.currentTimeMillis();
        if (now - lastSpawnTime < spawnInterval || powerUps.size() >= maxOnMap) return;
        lastSpawnTime = now;

        // El panel puede medir 0 antes de mostrarse
        if (mapWidth <= PowerUp.SIZE || mapHeight <= PowerUp.SIZE) return;

        double x = random.nextInt(mapWidth - PowerUp.SIZE);
        double y = random.nextInt(mapHeight - PowerUp.SIZE);
        powerUps.add(createRandom(x, y));
    }

    private PowerUp createRandom(double x, double y) {
        return switch (random.nextInt(5)) {
            case 0 -> new Heal(x, y, 25);
            case 1 -> new Shield(x, y, 50);
            case 2 -> new DamageBoost(x, y, 10000);
            case 3 -> new Ammo(x, y, 30);
            default -> new MaxAmmo(x, y);
        };
    }
}

            this.actualMunition--;
            this.lastShoot = currentTime;
 
            for (int i = 0; i < 4; i++) {
                // random.nextDouble() da un valor entre 0.0 y 1.0.

                double offset = (random.nextDouble() * 0.5) * 0.6;

                double dirY = Math.sin(spreadAngle);
 
                double muzzleX = getMuzzleX(player);
                double muzzleY = getMuzzleY(player);
 
                Bullet pellet = new Bullet(muzzleX, muzzleY, 6, 6, dirX, dirY, this.getDamage(), 30, BulletKind.PELLET);
                state.addPlayerBullet(pellet);
            }
            return true;
        }
        return false;
    }

    @Override 
    public boolean isAutomatic() {
        return false;
    }
}

public class WeaponItem extends Entity {

    private final FireArm weapon;

    public WeaponItem(double x, double y, FireArm weapon) {
        super(x, y, 48, 32);
        this.weapon = weapon;
    }

    public FireArm getWeapon() {
        return weapon;
    }
}
