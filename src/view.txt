package view;

import java.awt.image.BufferedImage;

public class Animation {
    private final BufferedImage[] frames;
    private int currentFrame;
    private long lastTime;
    private final int delay; // Milisegundos entre cada fotograma

    public Animation(int delay, BufferedImage... frames) {
        this.delay = delay;
        this.frames = frames;
        this.lastTime = System.currentTimeMillis();
    }

    public void update() {
        long timer = System.currentTimeMillis() - this.lastTime;
        if (timer > this.delay) {
            this.currentFrame ++;
            if (this.currentFrame >= this.frames.length) {
                this.currentFrame = 0; // Vuelve al inicio para hacer un bucle
            }
            lastTime = System.currentTimeMillis();
        }
    }

    public BufferedImage getCurrentFrame() {
        return this.frames[this.currentFrame];
    }
}

package view;
 
import model.Bullet;

import java.awt.BasicStroke;

import java.awt.Stroke;

/**
 * Efectos visuales de disparo: estelas (tracers) de las balas y destello en la boca del arma.
 * Solo dibuja; no modifica el modelo. Hay que llamarlo con el Graphics2D en coordenadas de MUNDO.
 */
public class BulletEffects {
 
    private static final long FLASH_MS = 70;
    private static final long SHOTGUN_FLASH_MS = 100;
 
    // ESTELAS
 
    public void drawTracers(Graphics2D g2, List<Bullet> bullets) {
        Stroke oldStroke = g2.getStroke();
        Paint oldPaint = g2.getPaint();
 
        for (Bullet b : bullets) {
            if (b.isActive()) {
                drawTracer(g2, b);
            }
        }
 
        g2.setStroke(oldStroke);
        g2.setPaint(oldPaint);
    }
 
    private void drawTracer(Graphics2D g2, Bullet b) {
        boolean pellet = b.getKind() == BulletKind.PELLET;
 
        double maxLength = pellet ? 18 : 36;
        // La estela crece desde 0 los primeros ticks: parece que "sale" del arma
        double length = Math.min(maxLength, b.getAge() * b.getSpeed() + 6);
 
        double headX = b.getX() + b.getWidth() / 2.0;
        double headY = b.getY() + b.getHeight() / 2.0;
        double tailX = headX - b.getDirX() * length;
        double tailY = headY - b.getDirY() * length;
 
        Color glow = pellet ? new Color(255, 110, 20) : new Color(255, 185, 60);
        Line2D line = new Line2D.Double(tailX, tailY, headX, headY);
 
        // Halo exterior
        g2.setPaint(new GradientPaint((float) tailX, (float) tailY, alpha(glow, 0),
                (float) headX, (float) headY, alpha(glow, 90)));
        g2.setStroke(new BasicStroke(pellet ? 7f : 9f, BasicStroke.CAP_ROUND, BasicStroke.JOIN_ROUND));
        g2.draw(line);
 
        // Cuerpo
        g2.setPaint(new GradientPaint((float) tailX, (float) tailY, alpha(glow, 0),
                (float) headX, (float) headY, alpha(glow, 210)));
        g2.setStroke(new BasicStroke(4f, BasicStroke.CAP_ROUND, BasicStroke.JOIN_ROUND));
        g2.draw(line);
 
        // Núcleo blanco-amarillento
        Color core = new Color(255, 250, 210);
        g2.setPaint(new GradientPaint((float) tailX, (float) tailY, alpha(core, 0),
                (float) headX, (float) headY, alpha(core, 255)));
        g2.setStroke(new BasicStroke(2f, BasicStroke.CAP_ROUND, BasicStroke.JOIN_ROUND));
        g2.draw(line);
 
        // Punta brillante
        g2.setPaint(alpha(core, 235));
        g2.fill(new Ellipse2D.Double(headX - 2.5, headY - 2.5, 5, 5));
    }
 
    // DESTELLO DE BOCA DE FUEGO
 
    public void drawMuzzleFlash(Graphics2D g2, Player player) {
        if (!player.isAlive()) return;
 
        Weapon weapon = player.getActiveWeapon();
        if (!(weapon instanceof FireArm)) return; // el cuchillo no tiene destello
 
        boolean shotgun = weapon instanceof ShotGun;
        long duration = shotgun ? SHOTGUN_FLASH_MS : FLASH_MS;
        long elapsed = System.currentTimeMillis() - weapon.getLastShoot();
        if (elapsed < 0 || elapsed > duration) return;
 
        double life = 1.0 - (double) elapsed / duration; // 1 al disparar -> 0 al apagarse
 
        double centerX = player.getX() + player.getWidth() / 2.0;
        double centerY = player.getY() + player.getHeight() / 2.0;
 
        AffineTransform old = g2.getTransform();
        Paint oldPaint = g2.getPaint();
        Stroke oldStroke = g2.getStroke();
 
        // A partir de acá, el eje X apunta hacia donde mira el jugador y (0,0) es la boca del arma
        g2.translate(centerX, centerY);
        g2.rotate(player.getAngle());
        g2.translate(Weapon.MUZZLE_FORWARD, Weapon.MUZZLE_SIDE);
 
        // Misma semilla durante todo el destello: la forma se mantiene y solo se desvanece
        Random rnd = new Random(weapon.getLastShoot());
 
        drawFlashGlow(g2, life, shotgun);
        drawFlashSpikes(g2, rnd, life, shotgun);
        drawFlashSparks(g2, rnd, life, shotgun);
 
        g2.setTransform(old);
        g2.setPaint(oldPaint);
        g2.setStroke(oldStroke);
    }
 
    /** Resplandor redondo + luz tenue que "ilumina" la zona oscura alrededor. */
    private void drawFlashGlow(Graphics2D g2, double life, boolean shotgun) {
        Point2D center = new Point2D.Double(0, 0);
        float[] fractions = {0f, 0.35f, 1f};
 
        // Iluminación ambiente
        double ambient = shotgun ? 190 : 140;
        g2.setPaint(new RadialGradientPaint(center, (float) ambient, fractions, new Color[]{
                new Color(255, 170, 70, (int) (70 * life)),
                new Color(255, 120, 30, (int) (35 * life)),
                new Color(255, 100, 0, 0)}));
        g2.fill(new Ellipse2D.Double(-ambient, -ambient, ambient * 2, ambient * 2));
 
        // Núcleo brillante
        double radius = (shotgun ? 70 : 50) * (0.6 + 0.4 * life);
        g2.setPaint(new RadialGradientPaint(center, (float) radius, fractions, new Color[]{
                new Color(255, 245, 190, (int) (230 * life)),
                new Color(255, 150, 40, (int) (150 * life)),
                new Color(255, 100, 0, 0)}));
        g2.fill(new Ellipse2D.Double(-radius, -radius, radius * 2, radius * 2));
    }
 
    /** Picos de fuego en abanico hacia adelante. */
    private void drawFlashSpikes(Graphics2D g2, Random rnd, double life, boolean shotgun) {
        int count = shotgun ? 9 : 6;
        double spread = shotgun ? 0.75 : 0.5;     // apertura del abanico (rad)
        double baseLength = shotgun ? 70 : 48;
 
        Path2D outer = new Path2D.Double();
        Path2D inner = new Path2D.Double();
 
        for (int i = 0; i < count; i++) {
            // El primer pico va derecho (el más largo); el resto se reparte en abanico
            double angle = (i == 0) ? 0 : (rnd.nextDouble() * 2 - 1) * spread;
            double length = baseLength * (i == 0 ? 1.2 : 0.45 + rnd.nextDouble() * 0.6) * (0.5 + 0.5 * life);
            double halfWidth = (i == 0 ? 7 : 4) * (0.6 + 0.4 * life);
 
            addSpike(outer, angle, length, halfWidth);
            addSpike(inner, angle, length * 0.55, halfWidth * 0.5);
        }
 
        g2.setPaint(new Color(255, 160, 40, (int) (230 * life)));
        g2.fill(outer);
        g2.setPaint(new Color(255, 250, 215, (int) (255 * life)));
        g2.fill(inner);
    }
 
    /** Triángulo fino desde la boca del arma hacia el ángulo indicado. */
    private void addSpike(Path2D path, double angle, double length, double halfWidth) {
        double dx = Math.cos(angle);
        double dy = Math.sin(angle);
        double px = -dy * halfWidth; // perpendicular
        double py = dx * halfWidth;
 
        path.moveTo(px, py);
        path.lineTo(dx * length, dy * length);
        path.lineTo(-px, -py);
        path.closePath();
    }
 
    /** Chispas chicas que salen disparadas y se alejan mientras el destello se apaga. */
    private void drawFlashSparks(Graphics2D g2, Random rnd, double life, boolean shotgun) {
        int count = shotgun ? 9 : 5;
        double progress = 1.0 - life; // 0 -> 1
 
        g2.setStroke(new BasicStroke(1.6f, BasicStroke.CAP_ROUND, BasicStroke.JOIN_ROUND));
 
        for (int i = 0; i < count; i++) {
            double angle = (rnd.nextDouble() * 2 - 1) * 0.9;
            double maxDist = 30 + rnd.nextDouble() * (shotgun ? 70 : 45);
            double start = 6 + progress * maxDist;
            double end = start + 5 + rnd.nextDouble() * 7;
 
            g2.setPaint(new Color(255, 215, 120, (int) (255 * life)));
            g2.draw(new Line2D.Double(
                    Math.cos(angle) * start, Math.sin(angle) * start,
                    Math.cos(angle) * end, Math.sin(angle) * end));
        }
    }
 
    private static Color alpha(Color c, int a) {
        return new Color(c.getRed(), c.getGreen(), c.getBlue(), Math.max(0, Math.min(255, a)));
    }
}

    @Override
    protected void paintComponent(Graphics g) {
        super.paintComponent(g);
        Graphics2D g2 = (Graphics2D) g;

        g2.setRenderingHint(RenderingHints.KEY_ANTIALIASING, RenderingHints.VALUE_ANTIALIAS_ON);
        g2.setRenderingHint(RenderingHints.KEY_INTERPOLATION, RenderingHints.VALUE_INTERPOLATION_BILINEAR);

        Player player = this.controller.getState().getPlayer();

        ///////////////////////////////////////
        // SEGUIMIENTO DE CAMARA A PERSONAJE //
        ///////////////////////////////////////

        int screenWidth = getWidth();
        int screenHeight = getHeight();

        int mapWidth = this.controller.getState().getMapWidth();
        int mapHeight = this.controller.getState().getMapHeight();

        // Calculo de camara
        int camX = (int) player.getX() - (screenWidth / 2) + (player.getWidth() / 2);
        int camY = (int) player.getY() - (screenHeight / 2) + (player.getHeight() / 2);

        // Calculo para no irse del mapa
        camX = Math.max(0, Math.min(camX, mapWidth - screenWidth));
        camY = Math.max(0, Math.min(camY, mapHeight - screenHeight));

        g2.translate(-camX, -camY);

        if (backgroundForest != null) {
            g2.drawImage(backgroundForest, 0, 0, mapWidth, mapHeight, null);
        }

        // Dibuja PowerUps
        drawPowerUps(g2); // [POWER-UP] primero, para que queden debajo de zombies y jugador

        // Dibuja Armas en el suelo
        drawGroundWeapons(g2); // [ARMAS] segundo, para que queden debajo de zombies y jugador

        // Dibuja a los enemigos
        if (zombieWalkAnimation != null) {
            zombieWalkAnimation.update();
        }

        //Dibuja Zombie Tanque
        if (tankWalkAnimation != null) {
            tankWalkAnimation.update();
        }

        for (Animation anim : zombieAnimations.values()) {
            anim.update();
        }
        for (Enemy zombie : this.controller.getState().getEnemies()) {

            if (zombie.isAlive()) {
                AffineTransform transformOriginal = g2.getTransform();
                double centerX = zombie.getX() + (zombie.getWidth() / 2.0);
                double centerY = zombie.getY() + (zombie.getHeight() / 2.0);

                g2.translate(centerX, centerY);
                g2.rotate(zombie.getAngle());

                int w = zombie.getWidth();
                int h = zombie.getHeight();

                Animation anim = zombieAnimations.get(zombie.getClass());
                BufferedImage idle = zombieIdleSprites.get(zombie.getClass());

                if (anim != null && anim.getCurrentFrame() != null) {
                    g2.drawImage(anim.getCurrentFrame(), -w / 2, -h / 2, w, h, null);
                } else if (idle != null) {
                    g2.drawImage(idle, -w / 2, -h / 2, w, h, null);
                } else {
                    g2.setColor(Color.RED);
                    g2.fillOval(-w / 2, -h / 2, w, h);
                }

                g2.setTransform(transformOriginal);
                g2.setColor(Color.WHITE);
                g2.setFont(new Font("Monospaced", Font.BOLD, 20));
                String lifeText = "HP: " + zombie.getHealth();

                int textX = (int) zombie.getX();
                int textY = (int) zombie.getY();
                g2.drawString(lifeText, textX, textY);
            }
        }
        ///////////////////////////////////////
        //   Dibuja las balas del jugador    //
        ///////////////////////////////////////

        /*g2.setColor(Color.ORANGE);
        for (Bullet bullet : this.controller.getState().getPlayerBullets()) {
            g2.fillOval((int) bullet.getX(), (int) bullet.getY(), bullet.getWidth() - 2, bullet.getHeight() - 2);
        }*/

        ///////////////////////////////////////
        //   Dibuja las balas venenosas      //
        ///////////////////////////////////////

        g2.setColor(new Color(50, 205, 50));
        for (Bullet poison : this.controller.getState().getPoisonBullets()) {
            g2.fillOval((int) poison.getX(), (int) poison.getY(), poison.getWidth(), poison.getHeight());
        }

        ///////////////////////////////////////
        //          Dibuja al jugador        //
        ///////////////////////////////////////
        if (controller.getIsAlive()) {
            AffineTransform transformOriginal = g2.getTransform();

            double centerX = controller.getPlayerX() + (player.getWidth() / 2.0);
            double centerY = player.getY() + (player.getHeight() / 2.0);

            g2.translate(centerX, centerY);
            g2.rotate(player.getAngle());

            int w = player.getWidth();
            int h = player.getHeight();

            if (player.isMoving() && playerFeetAnimation != null && playerFeetAnimation.getCurrentFrame() != null) {
                BufferedImage feetFrame = playerFeetAnimation.getCurrentFrame();
                AffineTransform transformFeet = g2.getTransform();
                g2.scale(0.6, 0.6);
                g2.drawImage(feetFrame, -w / 2, -h / 2, w, h, null);
                g2.setTransform(transformFeet);
            }

            Weapon activeWeapon = player.getActiveWeapon();
            Animation currentAnim = playerAnimations.get(activeWeapon.getClass());

            if (currentAnim != null) {
                currentAnim.update();
                if (currentAnim.getCurrentFrame() != null) {
                    BufferedImage bodyFrame = currentAnim.getCurrentFrame();
                    g2.drawImage(bodyFrame, -w / 2, -h / 2, w, h, null);
                }
            } else {
                g2.setColor(Color.RED);
                g2.fillOval(-w / 2, -h / 2, w, h);
            }

            g2.setTransform(transformOriginal);
        }

        // Cursor de prueba
        setCursor(new Cursor(Cursor.CROSSHAIR_CURSOR));
        if (firstLayer != null) {
            g2.drawImage(firstLayer, 0, 0, mapWidth, mapHeight, null);
        }

        ///////////////////////////////////////

    private void drawFloating(Graphics2D g2, Entity e, Color glowColor, Consumer<Graphics2D> content) {
        double t = System.currentTimeMillis() / 1000.0;
        double phase = e.getX() * 0.05;                               // desfase por objeto
        double bob = Math.sin(t * 3.0 + phase) * 4;                   // sube y baja 4 px
        double scale = 1.0 + Math.sin(t * 4.0 + phase) * 0.06;        // pulso de +-6%
        int glowAlpha = (int) (70 + Math.sin(t * 4.0 + phase) * 40);  // brillo 30..110

        int w = e.getWidth();
        int h = e.getHeight();

        AffineTransform old = g2.getTransform();
        g2.translate(e.getX() + w / 2.0, e.getY() + h / 2.0 + bob);

        g2.setColor(new Color(glowColor.getRed(), glowColor.getGreen(), glowColor.getBlue(), glowAlpha));
        g2.fillOval(-w / 2 - 6, -h / 2 - 6, w + 12, h + 12);

        g2.scale(scale, scale);
        content.accept(g2);       // cada uno dibuja lo suyo, centrado en (0, 0)

        g2.setTransform(old);
    }

    private void drawPowerUps(Graphics2D g2) {
        for (PowerUp p : this.controller.getState().getPowerUps()) {
            BufferedImage sprite = powerUpSprites.get(p.getClass());
            int w = p.getWidth();
            int h = p.getHeight();

            drawFloating(g2, p, p.getColor(), g -> {
                if (sprite != null) {
                    g.drawImage(sprite, -w / 2, -h / 2, w, h, null);
                } else {
                    g.setColor(p.getColor());
                    g.fillOval(-w / 2, -h / 2, w, h);
                    g.setColor(Color.BLACK);
                    g.drawOval(-w / 2, -h / 2, w, h);
                    g.setFont(new Font("Monospaced", Font.BOLD, 11));
                    FontMetrics fm = g.getFontMetrics();
                    g.drawString(p.getLabel(), -fm.stringWidth(p.getLabel()) / 2, fm.getAscent() / 2 - 2);
                }
            });
        }
    }

    private void showFinalMessage(Graphics2D g2, String texto, Color color) {
        g2.setColor(new Color(0, 0, 0, 180)); // fondo semitransparente
        g2.fillRect(0, 0, getWidth(), getHeight());

        g2.setColor(color);
        g2.setFont(new Font("Monospaced", Font.BOLD, 40));
        FontMetrics fm = g2.getFontMetrics();
        int textX = (getWidth() - fm.stringWidth(texto)) / 2;
        int textY = getHeight() / 2;
        g2.drawString(texto, textX, textY);
    }

    private void drawGroundWeapons(Graphics2D g2) {
        for (WeaponItem item : this.controller.getState().getGroundWeapons()) {
            BufferedImage sprite = weaponSprites.get(item.getWeapon().getClass());
            int x = (int) item.getX();
            int y = (int) item.getY();
            int w = item.getWidth();
            int h = item.getHeight();

            if (sprite != null) {
                g2.drawImage(sprite, x, y, w, h, null);
            } else {
                g2.setColor(Color.LIGHT_GRAY);
                g2.fillRect(x, y, w, h);
                g2.setColor(Color.WHITE);
                g2.setFont(new Font("Monospaced", Font.PLAIN, 12));
                g2.drawString(item.getWeapon().getClass().getSimpleName(), x, y - 5);
            }
        }
    }

    private void drawBossIntro(Graphics2D g2) {
        int step = controller.getBossIntroStep();
        double p = controller.getBossIntroProgress();

        // el texto aparece y se desvanece de a poco
        double fade = Math.max(0.0, Math.min(1.0, Math.min(p / 0.25, (1.0 - p) / 0.25)));
        int alpha = (int) (255 * fade);

        // fondo oscuro
        g2.setColor(new Color(0, 0, 0, 215));
        g2.fillRect(0, 0, getWidth(), getHeight());

        String text;
        Color color;
        int size;
        if (step == 0) {
            text = "Felicidades, sobreviviste a la primer oleada, buen trabajo";
            color = new Color(190, 190, 190);
            size = 32;
        } else {
            text = "Preparate para el Jefe";
            color = new Color(190, 0, 0);
            size = 54;
        }

        // si el texto no entra en pantalla, se achica la letra
        Font font = new Font("Serif", Font.BOLD | Font.ITALIC, size);
        FontMetrics fm = g2.getFontMetrics(font);
        while (fm.stringWidth(text) > getWidth() - 80 && size > 12) {
            size--;
            font = new Font("Serif", Font.BOLD | Font.ITALIC, size);
            fm = g2.getFontMetrics(font);
        }
        g2.setFont(font);

        int x = (getWidth() - fm.stringWidth(text)) / 2;
        int y = getHeight() / 2;

        // temblor leve solo en "Preparate para el Jefe"
        if (step == 1) {
            x += (int) (Math.random() * 5) - 2;
            y += (int) (Math.random() * 5) - 2;
        }

        // sombra rojo oscuro y texto encima
        g2.setColor(new Color(70, 0, 0, alpha));
        g2.drawString(text, x + 3, y + 3);
        g2.setColor(new Color(color.getRed(), color.getGreen(), color.getBlue(), alpha));
        g2.drawString(text, x, y);

    }
}

package view;

import java.awt.*;
import java.awt.event.MouseListener;
import java.awt.event.MouseAdapter;
import java.awt.event.MouseEvent;

import javax.swing.JButton;

public class MenuButton extends JButton{

    private boolean hover;
    private static final Color NORMAL_BG_COLOR = new Color(15, 15, 15, 200);
    private static final Color HOVER_BG_COLOR = new Color(40, 10, 10, 230);
    private static final Color BORDER_COLOR = new Color(200, 0, 0);
    private static final Color NORMAL_TEXT_COLOR = new Color(200, 200, 200);
    private static final Color HOVER_TEXT_COLOR = Color.WHITE;

    public MenuButton(String text) {
        super(text);
        this.hover = false;

        setFont(new Font("Arial", Font.BOLD, 22));
        setForeground(NORMAL_TEXT_COLOR);

        setFocusPainted(false);
        setContentAreaFilled(false);
        setBorderPainted(false);
        setOpaque(false);
        setCursor(new Cursor(Cursor.CROSSHAIR_CURSOR));
        setPreferredSize(new Dimension(250, 50));
        setMaximumSize(new Dimension(250, 50));

        setAlignmentX(Component.CENTER_ALIGNMENT);

        addMouseListener((MouseListener) new MouseAdapter() {
            @Override 
            public void mouseEntered(MouseEvent e) {
                hover = true;
                setForeground(HOVER_TEXT_COLOR);
                repaint();
            }

            @Override 
            public void mouseExited(MouseEvent e) {
                hover = false;
                setForeground(NORMAL_TEXT_COLOR);
                repaint();
            }
        });
    }

    @Override 
    protected void paintComponent(Graphics g) {
        Graphics2D g2 = (Graphics2D) g.create();
        g2.setRenderingHint(RenderingHints.KEY_ANTIALIASING, RenderingHints.VALUE_ANTIALIAS_ON);

        int width = getWidth();
        int height = getHeight();
        // Fondo del botón
        g2.setColor(hover ? HOVER_BG_COLOR : NORMAL_BG_COLOR);
        g2.fillRect(2, 2, width - 4, height -4);
        // Borde exterior en rojo
        g2.setColor(BORDER_COLOR);
        // Si el mouse está encima, el borde se hace más grueso
        g2.setStroke(new BasicStroke(hover ? 3.0f : 1.0f));
        g2.drawRect(2, 2, width -4, height - 4);

        // Marco rojo tenue extra al hacer hover
        if (hover) {
            g2.setColor(new Color(255, 0, 0, 100));
            g2.setStroke(new BasicStroke(1.0f));
            g2.drawRect(6, 6, width - 12, height - 12);
        }

        g2.dispose();
        super.paintComponent(g);
    }
}

package view;

import javax.swing.*;

public class MenuPanel extends JPanel {
    
    private Image backgroundImage;

    private MenuButton optionButton;
    private MenuButton exitButton;

    private java.util.List<Ceniza> particulas = new java.util.ArrayList<>();
    private javax.swing.Timer particleTimer;

    private class Ceniza {
        int x, y, size, speed;
        
        public Ceniza(int width, int height) {
            x = (int) (Math.random() * width);
            y = (int) (Math.random() * height);
            size = (int) (Math.random() * 4) + 2; // Tamaño entre 2 y 5 píxeles
            speed = (int) (Math.random() * 3) + 1; // Velocidad de caída
        }
        
        public void update(int width, int height) {
            y += speed;
            x -= 1; // Viento suave hacia la izquierda
            
            // Si sale de la pantalla, reaparece arriba
            if (y > height) y = 0;
            if (x < 0) x = width;
        }
    }

    public MenuPanel() {

        setLayout(new BoxLayout(this, BoxLayout.Y_AXIS));
        setBorder(BorderFactory.createEmptyBorder(150, 50, 0, 0));

        java.net.URL imgUrl = getClass().getResource("/Resources/imagenes/fondo_menu_principal4.png");
        if (imgUrl != null) {
            this.backgroundImage = new ImageIcon(imgUrl).getImage();
        } else {
            System.err.println("No se encontró el fondo del menú principal");
        }
        
        JLabel titleLabel = new JLabel("NO WAY BACK");
        try {
            // Cargamos el archivo de la fuente
            java.io.InputStream fontStream = getClass().getResourceAsStream("/Resources/fonts/Nosifer-Regular.ttf");
            if (fontStream != null) {
                Font customFont = Font.createFont(Font.TRUETYPE_FONT, fontStream); //Nosifer-Regular.ttf

                // damos el estilo y el tamaño deseado (el 60f es un float que representa los píxeles)

                // aplicamos
                titleLabel.setForeground(new Color(149,6 ,6));
                titleLabel.setFont(titleFont);
                titleLabel.setAlignmentX(LEFT_ALIGNMENT);
            } else {
                System.err.println("No se encontró la fuente tipográfica");
            }

        } catch (Exception e) {
            System.out.println("Error al cargar la fuente, usando Arial por defecto.");
            titleLabel.setFont(new Font("Arial", Font.BOLD, 60));
        }   

        playButton = new MenuButton("Jugar");
        optionButton = new MenuButton("Opciones");
        exitButton = new MenuButton("Salir");

        playButton.setAlignmentX(Component.LEFT_ALIGNMENT);
        optionButton.setAlignmentX(Component.LEFT_ALIGNMENT);
        exitButton.setAlignmentX(Component.LEFT_ALIGNMENT);


        add(Box.createVerticalGlue());

        add(titleLabel);
        add(Box.createRigidArea(new Dimension(0, 80)));
        add(playButton);
        add(Box.createRigidArea(new Dimension(0, 20)));
        add(optionButton);
        add(Box.createRigidArea(new Dimension(0, 20)));
        add(exitButton);

        add(Box.createVerticalGlue());

        // Generar 70 partículas iniciales (asumiendo un ancho base de 1920x1080 para esparcirlas)
        for (int i = 0; i < 70; i++) {
            particulas.add(new Ceniza(1920, 1080));
        }

        // Bucle de animación (se actualiza cada 16ms = ~60 FPS)
        particleTimer = new javax.swing.Timer(16, e -> {
            int currentWidth = getWidth() > 0 ? getWidth() : 1920;
            int currentHeight = getHeight() > 0 ? getHeight() : 1080;
            
            for (Ceniza c : particulas) {
                c.update(currentWidth, currentHeight);
            }
            repaint(); // Obliga a redibujar el panel con las nuevas posiciones
        });
        particleTimer.start();

        
    }

            g.fillOval(c.x, c.y, c.size, c.size);
        }
    }
