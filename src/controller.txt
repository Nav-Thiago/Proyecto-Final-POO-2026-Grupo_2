package controller;

import javax.sound.sampled.*;
import java.io.IOException;

public class AudioManager {

    private static Clip bgmClip;
    private static FloatControl volumeControl;

    private static int currentVolume = 50;

    public static void playMusic(String filePath) {
        try {
            if (bgmClip != null && bgmClip.isRunning()) {
                return;
            }

            java.net.URL audioUrl = AudioManager.class.getResource(filePath);
            if (audioUrl == null) {
                System.err.println("No se encontró el archivo de audio: " + filePath);
                return;
            }

            AudioInputStream audioStream = AudioSystem.getAudioInputStream(audioUrl);

            bgmClip = AudioSystem.getClip();
            bgmClip.open(audioStream);

            volumeControl = (FloatControl) bgmClip.getControl(FloatControl.Type.MASTER_GAIN);
            setVolume(currentVolume);

            bgmClip.loop(Clip.LOOP_CONTINUOUSLY);
            bgmClip.start();

        } catch (UnsupportedAudioFileException | IOException | LineUnavailableException e) {
            System.err.println("Error al reproducir audio: " + e.getMessage());
        }
    }

    public static void stopMusic() {
        if (bgmClip != null && bgmClip.isRunning()) {
            bgmClip.stop();
        }
    }

    public static void setVolume(int percentage) {
        if (volumeControl != null) {
            if (percentage <= 0) {
                volumeControl.setValue(volumeControl.getMinimum());
            } else {
                float dB = (float) (Math.log10((double) percentage / 100.0) * 20.0);
                volumeControl.setValue(dB);
            }
        }
    }
}

package controller;

import model.GameModel;
import model.GameState;
import view.GamePanel;

import java.awt.event.KeyAdapter;
import java.awt.event.KeyEvent;
import java.awt.event.MouseEvent;
import java.awt.event.MouseListener;
import java.awt.event.MouseMotionListener; 

public class GameController extends KeyAdapter implements MouseMotionListener, MouseListener {

    private final GameModel model;
    private GamePanel gamePanel;

    public GameController(GameModel model) {
        this.model = model;
    }

    public void setGamePanel (GamePanel gamePanel) {
        this.gamePanel = gamePanel;
        this.gamePanel.addKeyListener(this);
        this.gamePanel.addMouseListener(this);
        this.gamePanel.addMouseMotionListener(this);
    }

    // TECLADO
    @Override
    public void keyPressed(KeyEvent e) {
        switch (e.getKeyCode()) {
            case KeyEvent.VK_W, KeyEvent.VK_UP -> this.model.getState().getPlayer().setMoveUp(true);
            case KeyEvent.VK_S, KeyEvent.VK_DOWN -> this.model.getState().getPlayer().setMoveDown(true);
            case KeyEvent.VK_A, KeyEvent.VK_LEFT -> this.model.getState().getPlayer().setMoveLeft(true);
            case KeyEvent.VK_D, KeyEvent.VK_RIGHT -> this.model.getState().getPlayer().setMoveRight(true);
            case KeyEvent.VK_Q -> this.model.getState().getPlayer().swapWeapon();
            case KeyEvent.VK_E -> this.model.playerInteract();
            case KeyEvent.VK_ESCAPE -> this.model.togglePause();
            default -> {
            }
        }
    }

    @Override
    public void keyReleased(KeyEvent e) {
        switch (e.getKeyCode()) {
            case KeyEvent.VK_W, KeyEvent.VK_UP -> this.model.getState().getPlayer().setMoveUp(false);
            case KeyEvent.VK_S, KeyEvent.VK_DOWN -> this.model.getState().getPlayer().setMoveDown(false);
            case KeyEvent.VK_A, KeyEvent.VK_LEFT -> this.model.getState().getPlayer().setMoveLeft(false);
            case KeyEvent.VK_D, KeyEvent.VK_RIGHT -> this.model.getState().getPlayer().setMoveRight(false);
            default -> {}
        }
    }

    // MOUSE

    private void aimWithCamera(int screenmouseX, int screenmouseY) {
        model.Player player = this.model.getState().getPlayer();

        int screenWidth = gamePanel.getWidth();
        int screenHeight = gamePanel.getHeight();

        int mapWidth = this.model.getState().getMapWidth();
        int mapHeight = this.model.getState().getMapHeight();

        int camX = (int) player.getX() - (screenWidth / 2) + (player.getWidth() / 2);
        int camY = (int) player.getY() - (screenHeight / 2) + (player.getHeight() / 2);

        // Calculo para no irse del mapa
        camX = Math.max(0, Math.min(camX, mapWidth - screenWidth));
        camY = Math.max(0, Math.min(camY, mapHeight - screenHeight));

        double worldMouseX = screenmouseX + camX;
        double worldMouseY = screenmouseY + camY;

        player.aimAt(worldMouseX, worldMouseY);
    }

    @Override
    public void mouseMoved(MouseEvent e) {
        //this.model.getState().getPlayer().aimAt(e.getX(), e.getY());
        aimWithCamera(e.getX(), e.getY());
    }

    @Override
    // Manteniendo algún botón pulsado
    public void mouseDragged(MouseEvent e) {
        //this.model.getState().getPlayer().aimAt(e.getX(), e.getY());
        aimWithCamera(e.getX(), e.getY());
    }

    @Override
    public void mousePressed(MouseEvent e) {
        gamePanel.requestFocusInWindow();

        if (e.getButton() == MouseEvent.BUTTON1) {
            this.model.getState().getPlayer().attack(model.getState());
            this.model.getState().getPlayer().setAttacking(true);
        }
    }

    @Override
    public void mouseReleased(MouseEvent e) {

        if (e.getButton() == MouseEvent.BUTTON1) {
            this.model.getState().getPlayer().setAttacking(false);
        }
    }

    // Agrego debido a que tira error por el implements, se puede solucionar cambiando a:

        return this.model.getState();
    }

    public double getPlayerX() {
        return this.model.getState().getPlayer().getX();
    }

    public boolean getIsAlive() {
        return this.model.getState().getPlayer().isAlive();
    }

    public GameModel.gameStatus getStatus() {
        return this.model.getStatus();

===== GameLoopThread.java =====
package controller;

import model.GameModel;
import view.GamePanel;

import javax.swing.*;

public class GameLoopThread extends Thread {

    private static final int msPorFrame = 16; // 60 FPS -- 1000/60=16.66
    private final GamePanel gamePanel;
    private final GameModel model;

    public GameLoopThread(GamePanel gamePanel, GameModel model) {
        this.gamePanel = gamePanel;
        this.model = model;
        setDaemon(true); // Hilo de fondo para apagar el programa entero
    }

    @Override
    public void run() {  // Se redefine lo que debe hacer el hilo cuando se ejecute el .start()
        while (true) { // Game Loop
            model.update(gamePanel.getWidth(), gamePanel.getHeight());
            SwingUtilities.invokeLater(gamePanel :: repaint);
            try {
                Thread.sleep(msPorFrame); // Se pide al hilo que "descanse" por 16 ms para que el bucle no use el 100% del procesador
            } catch (InterruptedException e) {
                Thread.currentThread().interrupt(); // Maneja la excepción por si se cierra cuando esta "descansando"
                break;
            }
        }
    }
}

package controller;

import model.*;
import view.*;

import java.awt.CardLayout;

import javax.swing.JPanel;
import javax.swing.Timer;

public class MenuController implements MenuActions{

    private MainFrame mainFrame;
    private GameController controller;

    private CardLayout cardLayout;
    private JPanel mainContainer;

    public MenuController() {
        mainFrame = new MainFrame();
        cardLayout = new CardLayout();
        mainContainer = new JPanel(cardLayout);

        SplashUniversity uniSplash = new SplashUniversity();
        SplashSubject subjectSplash = new SplashSubject();
        MenuPanel menu = new MenuPanel();
        OptionsPanel option = new OptionsPanel(this);

        menu.getPlayButton().addActionListener(e -> startGame());
        menu.getOptionButton().addActionListener(e -> optionMenu());
        menu.getExitButton().addActionListener(e -> System.exit(0));
        

        mainContainer.add(uniSplash, "SPLASH_UNI");
        mainContainer.add(subjectSplash, "SPLASH_SUB");
        mainContainer.add(menu, "MENU");
        mainContainer.add(option, "OPTIONS");

        mainFrame.add(mainContainer);

        mainFrame.setVisible(true);

        // Loop de cards (Universidad -> Materia -> Menu)
        showUniversitySplash();
    }

    private void showUniversitySplash() {
        AudioManager.playMusic("/Resources/audio/soundtrack1.wav");
        AudioManager.setVolume(50);
        cardLayout.show(mainContainer, "SPLASH_UNI");

        Timer timer = new Timer(2000, e -> {
            showSubjectSplash();
        });
        timer.setRepeats(false);
        timer.start();
    }

    private void showSubjectSplash() {
        cardLayout.show(mainContainer, "SPLASH_SUB");

        Timer timer = new Timer(2000, e -> {
            showMainMenu();
        });
        timer.setRepeats(false);
        timer.start();
    }

    private void showMainMenu() {
        cardLayout.show(mainContainer, "MENU");
        //AudioManager.playMusic("Resources/audio/soundtrack1.wav");
        //AudioManager.setVolume(50);
    }

    // Implemento el método de la interfaz "MenuActions" para poder utilizarlo en "MenuPanel"
    @Override 
    public void backToMenu() {

    }

    private void showOptionsMenu() {
        cardLayout.show(mainContainer, "OPTIONS");
    }

    @Override 
    public void optionMenu() {
        showOptionsMenu();
    }

    @Override 
    public void startGame() {
        AudioManager.stopMusic();
        Player player = new Player(128, 3040);
        player.equipWeapon(new Handgun(20, 400, 20, 20));

        // Dimensiones del mapa 
        int mapCols = 100;
        int mapRows = 100;
        int widthPixels = mapCols * 32;
        int heightPixels = mapRows * 32;

        GameState state = new GameState(player, widthPixels, heightPixels);
        boolean[][] matrix = MapLoader.loadCollisionMatrix("/Resources/mapa_bosque/mapa_bosque_NWB.tmj", mapCols, mapRows);
        state.setCollisionMatrix(matrix);
        state.getGroundWeapons().add(new WeaponItem(1753, 1415, new ShotGun(30, 800, 7, 7)));
        state.getGroundWeapons().add(new WeaponItem(2807, 323, new Rifle(30, 200, 30, 30)));

        GameModel model = new GameModel(state);
        this.controller = new GameController(model);
        GamePanel gamePanel = new GamePanel(this.controller);
        this.controller.setGamePanel(gamePanel);

        //gamePanel.addKeyListener(controller);
        //gamePanel.addMouseMotionListener(controller);
        //gamePanel.addMouseListener(controller);

        mainContainer.add(gamePanel, "GAME");
        mainContainer.revalidate();
        cardLayout.show(mainContainer, "GAME");

        gamePanel.setFocusable(true);
        gamePanel.requestFocusInWindow();

        gamePanel.addComponentListener(new java.awt.event.ComponentAdapter() {
            @Override 
            public void componentShown(java.awt.event.ComponentEvent e) {
                gamePanel.requestFocusInWindow();
            }
        });

        GameLoopThread loopThread = new GameLoopThread(gamePanel, model);
        loopThread.start(); // Le pide al sistema operativo que cree un nuevo hilo de procesamiento real y que dentro de ese hilo ejecute el método run()
    }

        mainFrame.setSize(width, height);
        mainFrame.setLocationRelativeTo(null);
        mainContainer.revalidate();
