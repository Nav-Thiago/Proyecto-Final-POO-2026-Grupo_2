package controller;

import javax.sound.sampled.*;
import java.io.IOException;

public class AudioManager {

    private static Clip bgmClip;
    private static FloatControl volumeControl;

    private static int currentVolume = 50;

    public static void playMusic(String filePath) {
        try {
            if (bgmClip != null && bgmClip.isRunning()) {
                return;
            }

            java.net.URL audioUrl = AudioManager.class.getResource(filePath);
            if (audioUrl == null) {
                System.err.println("No se encontró el archivo de audio: " + filePath);
                return;
            }

            AudioInputStream audioStream = AudioSystem.getAudioInputStream(audioUrl);

            bgmClip = AudioSystem.getClip();
            bgmClip.open(audioStream);

            volumeControl = (FloatControl) bgmClip.getControl(FloatControl.Type.MASTER_GAIN);
            setVolume(currentVolume);

            bgmClip.loop(Clip.LOOP_CONTINUOUSLY);
            bgmClip.start();

        } catch (UnsupportedAudioFileException | IOException | LineUnavailableException e) {
            System.err.println("Error al reproducir audio: " + e.getMessage());
        }
    }

    public static void stopMusic() {
        if (bgmClip != null && bgmClip.isRunning()) {
            bgmClip.stop();
        }
    }

    public static void setVolume(int percentage) {
        if (volumeControl != null) {
            if (percentage <= 0) {
                volumeControl.setValue(volumeControl.getMinimum());
            } else {
                float dB = (float) (Math.log10((double) percentage / 100.0) * 20.0);
                volumeControl.setValue(dB);
            }
        }
    }
}

package controller;

import model.*;
import view.GamePanel;

import java.awt.event.KeyAdapter;
import java.awt.event.KeyEvent;
import java.awt.event.MouseEvent;
import java.awt.event.MouseListener;
import java.awt.event.MouseMotionListener;

public class GameController extends KeyAdapter implements MouseMotionListener, MouseListener {

    private final GameModel model;
    private GamePanel gamePanel;

    public GameController(GameModel model) {
        this.model = model;
    }

    public void setGamePanel (GamePanel gamePanel) {
        this.gamePanel = gamePanel;
        this.gamePanel.addKeyListener(this);
        this.gamePanel.addMouseListener(this);
        this.gamePanel.addMouseMotionListener(this);
    }

    // TECLADO
    @Override
    public void keyPressed(KeyEvent e) {
        switch (e.getKeyCode()) {
            case KeyEvent.VK_W, KeyEvent.VK_UP -> this.model.getState().getPlayer().setMoveUp(true);
            case KeyEvent.VK_S, KeyEvent.VK_DOWN -> this.model.getState().getPlayer().setMoveDown(true);
            case KeyEvent.VK_A, KeyEvent.VK_LEFT -> this.model.getState().getPlayer().setMoveLeft(true);
            case KeyEvent.VK_D, KeyEvent.VK_RIGHT -> this.model.getState().getPlayer().setMoveRight(true);
            case KeyEvent.VK_Q -> this.model.getState().getPlayer().swapWeapon();
            case KeyEvent.VK_E -> this.model.playerInteract();
            case KeyEvent.VK_ESCAPE -> this.model.togglePause();
            default -> {
            }
        }
    }

    @Override
    public void keyReleased(KeyEvent e) {
        switch (e.getKeyCode()) {
            case KeyEvent.VK_W, KeyEvent.VK_UP -> this.model.getState().getPlayer().setMoveUp(false);
            case KeyEvent.VK_S, KeyEvent.VK_DOWN -> this.model.getState().getPlayer().setMoveDown(false);
            case KeyEvent.VK_A, KeyEvent.VK_LEFT -> this.model.getState().getPlayer().setMoveLeft(false);
            case KeyEvent.VK_D, KeyEvent.VK_RIGHT -> this.model.getState().getPlayer().setMoveRight(false);
            default -> {}
        }
    }

    // MOUSE
    @Override
    public void mouseMoved(MouseEvent e) {
        //this.model.getState().getPlayer().aimAt(e.getX(), e.getY());
        aimWithCamera(e.getX(), e.getY());
    }

    @Override
    // Manteniendo algún botón pulsado
    public void mouseDragged(MouseEvent e) {
        //this.model.getState().getPlayer().aimAt(e.getX(), e.getY());
        aimWithCamera(e.getX(), e.getY());
    }

    @Override
    public void mousePressed(MouseEvent e) {
        gamePanel.requestFocusInWindow();

        if (e.getButton() == MouseEvent.BUTTON1) {
            this.model.getState().getPlayer().attack(model.getState());
            this.model.getState().getPlayer().setAttacking(true);
        }

        if (e.getButton() == MouseEvent.BUTTON3) {
            System.out.println("X: " + model.getState().getPlayer().getX() + "| Y: " + model.getState().getPlayer().getY());
        }
    }

    @Override
    public void mouseReleased(MouseEvent e) {
        if (e.getButton() == MouseEvent.BUTTON1) {
            this.model.getState().getPlayer().setAttacking(false);
        }
    }

    private void aimWithCamera(int screenmouseX, int screenmouseY) {
        model.Player player = this.model.getState().getPlayer();

        int screenWidth = gamePanel.getWidth();
        int screenHeight = gamePanel.getHeight();

        int mapWidth = this.model.getState().getMapWidth();
        int mapHeight = this.model.getState().getMapHeight();

        int camX = calculateCameraX(player, screenWidth, mapWidth);
        int camY = calculateCameraY(player, screenHeight, mapHeight);

        double worldMouseX = screenmouseX + camX;
        double worldMouseY = screenmouseY + camY;

        player.aimAt(worldMouseX, worldMouseY);
    }

    private int calculateCameraX(Player player, int screenWidth, int mapWidth) {
        int camX = (int) player.getX() - (screenWidth / 2) + (player.getWidth() / 2);
        return Math.max(0, Math.min(camX, mapWidth - screenWidth));
    }

    private int calculateCameraY(Player player, int screenHeight, int mapHeight) {
        int camY = (int) player.getY() - (screenHeight / 2) + (player.getHeight() / 2);
        return Math.max(0, Math.min(camY, mapHeight - screenHeight));
    }

    public void drawScene(GamePanel panel) {
        GameState state = this.model.getState();
        Player player = state.getPlayer();

        int screenWidth = panel.getWidth();
        int screenHeight = panel.getHeight();
        int mapWidth = state.getMapWidth();
        int mapHeight = state.getMapHeight();

        // Cálculo de cámara
        int camX = calculateCameraX(player, screenWidth, mapWidth);
        int camY = calculateCameraY(player, screenHeight, mapHeight);

        panel.applyCamera(camX, camY);
        panel.drawBackground(mapWidth, mapHeight);

        // Dibuja al power-up
        drawPowerUps(panel, state); // [POWER-UP] primero, para que queden debajo de zombies y jugador

        // Dibuja Armas en el suelo
        drawGroundWeapons(panel, state); // [ARMAS] segundo, para que queden debajo de zombies y jugador

        // Dibuja los zombies
        drawZombies(panel, state);

        // Dibuja a las balas venenosas
        drawPoisonBullets(panel, state);

        // Dibuja al jugador
        drawPlayer(panel, state);

        // Dibuja la primera capa del mapa
        drawFirstLayer(panel, state);

        // Dibuja la oscuridad y el cono de visión
        drawDarkness(panel, state);

        // Dibuja los disparos de las armas
        drawBulletTracers(panel, state);
        drawMuzzleFlash(panel, state);

        // Deshace la cámara
        panel.resetCamera(camX, camY);

        // Dibuja el HUD
        drawHud(panel, state);

        // Dibuja los mensajes de estado
        drawStatusMessage(panel);
    }

    private void drawPowerUps(GamePanel panel, GameState state) {
        for (PowerUp p : state.getPowerUps()) {
            panel.drawPowerUp(p.getX(), p.getY(), p.getWidth(), p.getHeight(), p.getClass().getSimpleName());
        }
    }

    private void drawGroundWeapons(GamePanel panel, GameState state) {
        for (WeaponItem item : state.getGroundWeapons()) {
            panel.drawGroundWeapon(item.getX(), item.getY(), item.getWidth(), item.getHeight(), item.getWeapon().getClass().getSimpleName());
        }
    }

    private void drawZombies(GamePanel panel, GameState state) {
        panel.updateZombieAnimations();
        for (Enemy zombie : state.getEnemies()) {
            if (zombie.isAlive()) {
                // Le pasamos zombie.isAttacking() al final
                panel.drawZombie(zombie.getX(), zombie.getY(), zombie.getWidth(), zombie.getHeight(), zombie.getAngle(), zombie.getHealth(), zombie.getClass().getSimpleName(), zombie.isAttacking());
            }
        }
    }
    
    private void drawPoisonBullets(GamePanel panel, GameState state) {
        for (Bullet poison : state.getPoisonBullets()) {
            panel.drawPoisonBullet(poison.getX(), poison.getY(), poison.getWidth(), poison.getHeight());
        }
    }

    private void drawPlayer(GamePanel panel, GameState state) {
        Player player = state.getPlayer();
        if (player.isAlive()) {
            panel.drawPlayer(player.getX(), player.getY(), player.getWidth(), player.getHeight(), player.getAngle(), player.isMoving(), player.getActiveWeapon().getClass().getSimpleName());
        }
    }

    private void drawFirstLayer(GamePanel panel, GameState state) {
        panel.drawFirstLayer(state.getMapWidth(), state.getMapHeight());
    }

    private void drawDarkness(GamePanel panel, GameState state) {
        Player player = state.getPlayer();
        panel.drawDarkness(player.getX(), player.getY(), player.getWidth(), player.getHeight(), player.getAngle(), state.getMapWidth(), state.getMapHeight());
    }

    private void drawHud(GamePanel panel, GameState state) {
        Player player = state.getPlayer();

        panel.drawHealth(player.getHealth(), player.getMaxHealth());

        if (player.getShield() > 0) {
            panel.drawShield(player.getShield());
        }

        if (player.getDamageBonus() > 0) {
            panel.drawDamageBonus(player.getDamageBonus());
        }

        Weapon weapon = player.getActiveWeapon();
        if (weapon.isRanged()) {
            panel.drawFireArmInfo(weapon.getClass().getSimpleName(), weapon.getActualMunition(), weapon.getMaxMunition());
        } else {
            panel.drawMeleeInfo();
        }
    }

    private void drawStatusMessage(GamePanel panel) {
        GameModel.GameStatus status = this.model.getStatus();

        if (status == GameModel.GameStatus.GAME_OVER) {
            panel.drawGameOver(model.getMatchTimeMs(), model.getBestTimeMs(), model.isNewRecord());
        } else if (status == GameModel.GameStatus.GAME_WON) {
            panel.drawGameWon(model.getMatchTimeMs(), model.getBestTimeMs(), model.isNewRecord());
        } else if (status == GameModel.GameStatus.PAUSED) {
            panel.drawPause();
        } else if (status == GameModel.GameStatus.BOSS_INTRO) {
            panel.drawBossIntro(this.model.getBossIntroStep(), this.model.getBossIntroProgress());
        }
    }

        for (Bullet bullet : state.getPlayerBullets()) {

        }
    }

package controller;

import model.GameModel;
import view.GamePanel;

import javax.swing.*;

public class GameLoopThread extends Thread {

    private static final int msPorFrame = 16; // 60 FPS -- 1000/60=16.66
    private final GamePanel gamePanel;
    private final GameModel model;

    public GameLoopThread(GamePanel gamePanel, GameModel model) {
        this.gamePanel = gamePanel;
        this.model = model;
        setDaemon(true); // Hilo de fondo para apagar el programa entero
    }

    @Override
    public void run() {  // Se redefine lo que debe hacer el hilo cuando se ejecute el .start()
        while (true) { // Game Loop
            model.update();
            SwingUtilities.invokeLater(gamePanel :: repaint);
            try {
                Thread.sleep(msPorFrame); // Se pide al hilo que "descanse" por 16 ms para que el bucle no use el 100% del procesador
            } catch (InterruptedException e) {
                Thread.currentThread().interrupt(); // Maneja la excepción por si se cierra cuando esta "descansando"
                break;
            }
        }
    }
}

package model;

public class Bullet extends Entity {
    private final double dirX;
    private final double dirY;
    private final double speed;
    private final double damage;
    private final BulletKind kind;
    private boolean active; // booleano para saber si la bala debe 'desaparecer'
    private int lifeSpanTicks;
    private int age;        // ticks que lleva viva (la vista lo usa para que la estela "salga" del arma)

    public Bullet(double x, double y, int width, int height, double dirX, double dirY, double damage, int lifeSpanTicks) {
        this(x, y, width, height, dirX, dirY, damage, lifeSpanTicks, BulletKind.STANDARD);
    }

    public Bullet(double x, double y, int width, int height, double dirX, double dirY, double damage, int lifeSpanTicks, BulletKind kind) {
        super(x, y, width, height);
        this.dirX = dirX;
        this.dirY = dirY;
        this.speed = 12.0;
        this.damage = damage;
        this.active = true;
        this.lifeSpanTicks = lifeSpanTicks;
        this.kind = kind;
        this.age = 0;
    }

    public void update(int mapWidth, int mapHeight) {
        this.x += dirX * speed; 
        this.y += dirY * speed;

        this.age++;
        this.lifeSpanTicks--;

        if (this.lifeSpanTicks <= 0) {
            this.active = false;
        }

        // Desactivar la bala si sale de los límites de la ventana
        if (x < 0 || x > mapWidth || y < 0 || y > mapHeight) {
            active = false;
        }
    }

    public boolean isActive() {
        return active;
    }
    public void setActive(boolean active) {
        this.active = active;
    }
    public double getDamage(){
        return damage;
    }
    public double getDirX() {
        return dirX;
    }
    public double getDirY() {
        return dirY;
    }
    public BulletKind getKind() {
        return kind;
    }
    public int getAge() {
        return age;
    }
    public double getSpeed() {
        return speed;
    }
}

package model;

public abstract class Character extends Entity {

    protected double speed;
    private int health;
    private int maxHealth;

    public Character(double x, double y, int width, int height, double speed, int health, int maxHealth) {
        super(x, y, width, height);
        this.speed = speed;
        this.maxHealth = maxHealth;
        this.health = health;
    }

    public int getHealth() {
        return health;
    }
    public void setHealth(int health) {
        this.health = health;
    }

    public int getMaxHealth() {
        return maxHealth;
    }

    protected void move(double dx, double dy, GameState state) {
        if (dx != 0 || dy != 0) {
            double length = Math.sqrt(dx * dx + dy * dy);
            double moveX = (dx / length) * speed;
            double moveY = (dy / length) * speed;

            if (!isCollidingWithMap(x + moveX, y, state)) x += moveX;
            if (!isCollidingWithMap(x, y + moveY, state)) y += moveY;
        }
    }

    protected boolean isCollidingWithMap(double nextX, double nextY, GameState state) {
        boolean[][] map = state.getCollisionMatrix();
        if (map == null) return false;

        int tile_size = GameState.TILE_SIZE;

        // Hitbox lógica (más chica que sprite visual)
        int leftTile = (int) (nextX + 15) / tile_size;
        int rightTile = (int) (nextX + width - 15) / tile_size;
        int topTile = (int) (nextY + 30) / tile_size;
        int bottomTile = (int) (nextY + height - 5) / tile_size;

        if (leftTile < 0 || rightTile >= map.length || topTile < 0 || bottomTile >= map[0].length) {
            return true;
        }

        return map[leftTile][topTile] || map[rightTile][topTile] ||
           map[leftTile][bottomTile] || map[rightTile][bottomTile];
    }

    protected boolean canMove(double dx, double dy, GameState state) {
        if (dx == 0 && dy == 0) return false;

        double length = Math.hypot(dx, dy);
        double moveX = (dx / length) * speed;
        double moveY = (dy / length) * speed;

        return !isCollidingWithMap(x + moveX, y + moveY, state);
    }

    public double getSpeed() {
        return speed;
    }

    public void setSpeed(double speed) {
        this.speed = speed;
    }

    public void takeDamage(int amount) {
        if (amount < this.health) {
            health -= amount;
        } else {
            this.health = 0;
        }
    }

    public boolean isAlive() {
        return health > 0;
    }
}

        checkZombiePlayerCollision(state);
    }

    private void checkBulletZombieCollision (GameState state) {
        for (Bullet bullet : state.getPlayerBullets()) {
            if (!bullet.isActive()) continue;

            for (Enemy zombie : state.getEnemies()) {
                if (zombie.isAlive() && bullet.collidesWith(zombie)) {
                    bullet.setActive(false);

                    int healthBefore = zombie.getHealth();
                    zombie.takeDamage((int) bullet.getDamage());
                    state.getMatchStats().addDamageDealt(healthBefore - zombie.getHealth());
                    break;
                }
            }
        }
    }

    private void checkBulletMapCollision (GameState state) {
        boolean[][] map = state.getCollisionMatrix();
        if (map == null) return;

        int tileSize = GameState.TILE_SIZE;

        for (Bullet bullet : state.getPlayerBullets()) {
            if (!bullet.isActive()) continue;

            int tileX = (int) (bullet.getX() + bullet.getWidth() / 2.0) / tileSize;
            int tileY = (int) (bullet.getY() + bullet.getHeight() / 2.0) / tileSize;

            if (tileX >= 0 && tileX < map.length && tileY >= 0 && tileY < map[0].length) {
                if (map[tileX][tileY]) {
                    bullet.setActive(false);
                }
            }
        }

        for (PoisonBullet poison : state.getPoisonBullets()) {
            if (!poison.isActive()) continue;

            int tileX = (int) (poison.getX() + poison.getWidth() / 2.0) / tileSize;
            int tileY = (int) (poison.getY() + poison.getHeight() / 2.0) / tileSize;

            if (tileX >= 0 && tileX < map.length && tileY >= 0 && tileY < map[0].length) {
                if (map[tileX][tileY]) {
                    poison.setActive(false);
                }
            }
        }
    }

    private void checkPoisonPlayerCollision (GameState state) {
        Player player = state.getPlayer();
        if (!player.isAlive()) return;

        for (PoisonBullet poison : state.getPoisonBullets()) {
            if (poison.isActive() && poison.collidesWith(player)) {
                poison.setActive(false);
                poison.applyTo(player);//POLIMORFISMO: cada bala sabe qué hacer
            }
        }
    }

    private void checkZombiePlayerCollision(GameState state) {
        Player player = state.getPlayer();
        if (!player.isAlive()) return;

        for (Enemy zombie : state.getEnemies()) {
            if (!zombie.isAlive()) continue;

            if (zombie.collidesWith(player)) {
                // 1. Si es el primer instante en que choca, inicia el temporizador del choque
                if (!zombie.isCollidingWithPlayer()) {
                    zombie.setCollidingWithPlayer(true);
                    zombie.setCollisionStartTime(System.currentTimeMillis());
                }

                // 2. Debe tener su intervalo de ataque listo Y haber mantenido el choque por 400ms
                if (zombie.canAttack() && zombie.hasCompletedAttackDelay()) {
                    player.takeDamage(zombie.attack());
                    zombie.resetAttackCooldown();
                }
            } else {
                // 3. Si el jugador se aleja antes de los 400ms, se reinicia el choque
                zombie.setCollidingWithPlayer(false);
            }
        }
    }
}

package model;

public abstract class Enemy extends Character {

    private final int ATTACK_COOLDOWN= 1000;
    protected double angle;
    private long lastAttackTime = 0;
    private static final double[] AVOID_ANGLES_DEG = {0, 25, -25, 50, -50, 75, -75, 100, -100};
    private boolean isCollidingWithPlayer = false;
    private long collisionStartTime = 0;
    private final int ATTACK_WINDUP = 400;

    public Enemy(double x, double y, int width, int height, double speed, int health, int maxHealth) {
        super(x, y, width, height, speed, health, maxHealth);
        this.angle = 0.0;
    }

    public void setX(double x) {
        this.x = x;
    }

    public void setY(double y) {
        this.y = y;
    }

    public double getAngle() {
        return angle;
    }

    public boolean isCollidingWithPlayer() {
        return isCollidingWithPlayer;
    }

    public void setCollidingWithPlayer(boolean colliding) {
        this.isCollidingWithPlayer = colliding;
    }

    public void setCollisionStartTime(long time) {
        this.collisionStartTime = time;
    }

    public boolean hasCompletedAttackDelay() {
        return (System.currentTimeMillis() - collisionStartTime >= ATTACK_WINDUP);
    }

    public void update(Player target, GameState state) {
        // Verifica si el zombie está muerto antes de moverse
        if (!isAlive()) return;

        double playerCenterX = target.getX() + (target.getWidth() / 2.0);
        double playerCenterY = target.getY() + (target.getHeight() / 2.0);
        double zombieCenterX = this.x + (this.width / 2.0);
        double zombieCenterY = this.y + (this.height / 2.0);

        double dx = playerCenterX - zombieCenterX;
        double dy = playerCenterY - zombieCenterY;

        double distance = Math.hypot(dx, dy);
        this.angle = Math.atan2(dy, dx);

        if (distance > 1000) {
            return;
        }

        if (distance > 30) {
            moveTowardsWithAvoidance(dx, dy, state);
        }

        avoidOverlap(state);
    }

    private void moveTowardsWithAvoidance(double dx, double dy, GameState state) {
        double baseAngle = Math.atan2(dy, dx);
        double length = Math.hypot(dx, dy);

        if (length == 0) return;

        for (double offsetDeg : AVOID_ANGLES_DEG) {
            double testAngle = baseAngle + Math.toRadians(offsetDeg);
            double testDx = Math.cos(testAngle) * length;
            double testDy = Math.sin(testAngle) * length;

            if (canMove(testDx, testDy, state)) {
                move(testDx, testDy, state);
                return;
            }
        }
    }

    public void avoidOverlap(GameState state) {
        for (Enemy otherZombie : state.getEnemies()) {
            if (this == otherZombie || !otherZombie.isAlive()) continue;

            if (this.collidesWith(otherZombie)) {

                double dx = this.getX() - otherZombie.getX();
                double dy = this.getY() - otherZombie.getY();
                double distance = Math.sqrt(dx * dx + dy * dy);

                if (distance > 0) {
                    double pushX = (dx/distance) * 1.5;
                    double pushY = (dy/distance) * 1.5;

                    if (!isCollidingWithMap(x + pushX, y, state)) {
                        x += pushX;
                    }

                    if (!isCollidingWithMap(x, y + pushY, state)) {
                        y += pushY;
                    }
                }
            }
        }
    }

    public boolean canAttack() {
        long currentTime = System.currentTimeMillis();
        return (currentTime - lastAttackTime >= ATTACK_COOLDOWN);
    }

    public void resetAttackCooldown() {
        this.lastAttackTime = System.currentTimeMillis();
    }

    public abstract int attack();

    public boolean isAttacking() {
        return (System.currentTimeMillis() - lastAttackTime < 500);
    }
}

package model;

import java.util.Random;

public class EnemySpawner {
    private long lastSpawnTime;
    private int spawnInterval;
    private int zombieSpawned;
    private final int waveLimit;

    public EnemySpawner(int spawnInterval, int waveLimit){
        this.spawnInterval = spawnInterval;
        this.waveLimit = waveLimit;
        this.lastSpawnTime = System.currentTimeMillis();
        this.zombieSpawned = 0;
    }
    
    public void update(GameState state){
        long currentTime = System.currentTimeMillis();

        if (currentTime - lastSpawnTime >= spawnInterval && zombieSpawned < waveLimit) {
            Random random = new Random();
            int tipoZombie = random.nextInt(4);

            int mapWidth = state.getMapWidth();
            int mapHeight = state.getMapHeight();
            boolean[][] collisionMatrix = state.getCollisionMatrix();

            if (mapWidth <= 100 || mapHeight <= 100 || collisionMatrix == null) return;

            int spawnX = 0;
            int spawnY = 0;
            boolean validPosition = false;
            int maxAttempts = 50; 
            int attempts = 0;
            int maxZombieWidth = 86;
            int maxZombieHeight = 86;

            while (!validPosition && attempts < maxAttempts) {
                spawnX = random.nextInt(mapWidth - maxZombieWidth);
                spawnY = random.nextInt(mapHeight - maxZombieHeight);
                if (!isOnBlockedTile(spawnX, spawnY, maxZombieWidth, maxZombieHeight, collisionMatrix)) {
                    validPosition = true;
                }
                attempts++;
            }

            Enemy newZombie;

            if (tipoZombie == 0) {
                newZombie = new TankZombie(spawnX, spawnY);
            } else if (tipoZombie == 1) {
                newZombie = new FastZombie(spawnX, spawnY);
            } else if (tipoZombie == 2) {
                newZombie = new PoisonZombie(spawnX, spawnY, state.getPoisonBullets());
            } else {
                newZombie = new NormalZombie(spawnX, spawnY);
            }

            state.addEnemy(newZombie);
            lastSpawnTime = currentTime;
            zombieSpawned++;
        }
    }

    // Método extraído de la lógica del PowerUpSpawner
    private boolean isOnBlockedTile(double x, double y, int width, int height, boolean[][] map) {
        int tile = GameState.TILE_SIZE;
        int fromX = (int) x / tile;
        int toX = (int) (x + width - 1) / tile;
        int fromY = (int) y / tile;
        int toY = (int) (y + height - 1) / tile;

        for (int tx = fromX; tx <= toX; tx++) {
            for (int ty = fromY; ty <= toY; ty++) {
                if (tx < 0 || tx >= map.length || ty < 0 || ty >= map[0].length) return true;
                if (map[tx][ty]) return true; // Hay colisión
            }
        }
        return false; // Área completamente libre
    }

    public boolean isWaveComplete(){
        return zombieSpawned >= waveLimit;
    }

    public void reset() {
        this.zombieSpawned = 0; 
        this.lastSpawnTime = System.currentTimeMillis();
    }
}

package model;

public abstract class FireArm extends Weapon implements AmmoHolder {

    protected int actualMunition;
    private final int maxMunition;

    public FireArm(int damage, long cooldown, int actualMunition, int maxMunition){
        super(damage);
        this.cooldown = cooldown;
        this.actualMunition = actualMunition;
        this.maxMunition = maxMunition;
    }

    @Override
    public boolean attack(Player player, GameState state){
        long currentTime = System.currentTimeMillis();

        if (this.actualMunition > 0 && currentTime - this.lastShoot >= this.cooldown) {

            this.actualMunition--;
            this.lastShoot = currentTime;

            double dirX = Math.cos(player.getAngle());
            double dirY = Math.sin(player.getAngle());

            double muzzleX = getMuzzleX(player);
            double muzzleY = getMuzzleY(player);

            double totalDamage = this.getDamage() + player.getDamageBonus();
            Bullet bullet = new Bullet(muzzleX, muzzleY, 6, 6, dirX, dirY, totalDamage, 60);

            state.addPlayerBullet(bullet);

            return true;
        }
        return false;
    }

    @Override
    public void addAmmo(int amount) {
        // El math.min asegura que no se pase de la munición máxima
        this.actualMunition = Math.min(this.actualMunition + amount, this.maxMunition);
    }

    @Override
    public void refillAmmo() {
        this.actualMunition = this.maxMunition;
    }

    @Override
    public boolean isRanged() {
        return true;
    }

    @Override
    public int getActualMunition() {
        return actualMunition;
    }

    @Override
    public int getMaxMunition() {
        return maxMunition;
    }
}

package model;

import java.time.LocalDate;
import java.util.List;

public class GameModel {

    private final GameState state;
    private final CollisionSystem collisionSystem;
    private final EnemySpawner spawner;
    private final PowerUpSpawner powerUpSpawner;

    public enum GameStatus {PLAYING, PAUSED, BOSS_INTRO, GAME_OVER, GAME_WON}
    private GameStatus status = GameStatus.PLAYING;
    private boolean bossSpawned = false;

    // Intro del jefe: 2 mensajes de INTRO_STEP_MS cada uno
    private static final int INTRO_STEPS = 2;
    private static final long INTRO_STEP_MS = 3500;
    private long bossIntroStart;

    private final MatchStats stats;
    private long lastUpdateTime = System.currentTimeMillis();

    private final MatchDAO matchDAO;

    private long matchTimeMs;
    private long bestTimeMs;
    private boolean newRecord;

    public GameModel(GameState state, MatchDAO matchDAO) {
        this.state = state;
        this.matchDAO = matchDAO;
        this.collisionSystem = new CollisionSystem();
        this.stats = state.getMatchStats();

        // Inicializamos los spawners
        this.spawner = new EnemySpawner(2000, 1);
        this.powerUpSpawner = new PowerUpSpawner(state, 4000, 20, 16); // cada 4 s, máx. 20 en el mapa, 16 al empezar

        //METODO PARA SPAWNEAR EL BOSS EN EL ESTACIONAMIENTO
        spawnBoss(6027, 5056);
    }

    public void update() {
        long now = System.currentTimeMillis();
        long delta = now - lastUpdateTime;
        lastUpdateTime = now;

        if (status == GameStatus.BOSS_INTRO) {
            updateBossIntro();
            return;
        }
        if (status != GameStatus.PLAYING) return;

        stats.addPlayedTime(delta);

        // Mover jugador
        Player player = state.getPlayer();
        if (!player.isAlive()) {
            saveMatch(false);
            status = GameStatus.GAME_OVER;
            return;
        }

        player.update(state);

        int mapWidth = state.getMapWidth();
        int mapHeight = state.getMapHeight();

        // Mover balas
        for (Bullet b : state.getPlayerBullets()) b.update(mapWidth, mapHeight);
        for (Bullet p: state.getPoisonBullets()) p.update(mapWidth, mapHeight);

        // Spawners
        spawner.update(state);
        powerUpSpawner.update();

        // Mover zombies
        for (Enemy zombie : state.getEnemies()) {
            if (zombie.isAlive()) {
                zombie.update(player, state);
            }
        }

        // Recolección de power-ups
        for (PowerUp p : state.getPowerUps()) {
            if (p.isActive() && player.collidesWith(p)) {
                p.collect(player);
            }
        }

        // Sistema de colisiones (daño)
        collisionSystem.update(state);

        // Los zombies muertos en este tick cuentan como eliminados
        for (Enemy zombie : state.getEnemies()) {
            if (!zombie.isAlive()) {
                stats.addKill();
            }
        }

        // Limpia los inactivos
        state.purgeInactive();

        // Oleada terminada -> intro del jefe; jefe derrotado -> fin de nivel
        if (state.getEnemies().isEmpty() && spawner.isWaveComplete()) {
            if (!bossSpawned) {
                startBossIntro();
            } else {
                saveMatch(true);
                status = GameStatus.GAME_WON;
            }
        }
    }

    private void startBossIntro() {
        status = GameStatus.BOSS_INTRO;
        bossIntroStart = System.currentTimeMillis();
    }

    private void updateBossIntro() {
        long elapsed = System.currentTimeMillis() - bossIntroStart;
        if (elapsed >= INTRO_STEP_MS * INTRO_STEPS) {
            spawnBoss(6027, 5056); 
            status = GameStatus.PLAYING;
        }
    }

    private void spawnBoss(double x, double y) {
        state.addEnemy(new FinalBoss(x, y, state.getPoisonBullets()));
        bossSpawned = true;
    }

    /** Qué mensaje toca mostrar: 0 = felicitación, 1 = "Preparate para el Jefe". */
    public int getBossIntroStep() {
        long elapsed = System.currentTimeMillis() - bossIntroStart;
        return (int) Math.min(INTRO_STEPS - 1, elapsed / INTRO_STEP_MS);
    }

    /** Avance (0.0 a 1.0) dentro del mensaje actual, para el efecto de aparecer y desvanecer. */
    public double getBossIntroProgress() {
        long elapsed = System.currentTimeMillis() - bossIntroStart;
        return Math.min(1.0, (elapsed % INTRO_STEP_MS) / (double) INTRO_STEP_MS);
    }

    public void togglePause() {
        if (status == GameStatus.PLAYING) {
            status = GameStatus.PAUSED;
        } else if (status == GameStatus.PAUSED) {
            status = GameStatus.PLAYING;
        }
    }

    public void playerInteract() {
        if (status != GameStatus.PLAYING) return;

        List<WeaponItem> groundWeapons = state.getGroundWeapons();
        Player player = state.getPlayer();

        for (int i = 0; i < groundWeapons.size(); i++) {
            WeaponItem item = groundWeapons.get(i);

            if (player.collidesWith(item)) {
                FireArm oldWeapon = player.getPrimaryWeapon();
                player.equipWeapon(item.getWeapon());
                groundWeapons.remove(i);

                if (oldWeapon != null) {
                    WeaponItem droppedWeapon = new WeaponItem(player.getX(), player.getY(), oldWeapon);
                    groundWeapons.add(droppedWeapon);
                }
                break;
            }
        }
    }

    public void saveMatch(boolean victory) {
        MatchRecord record = new MatchRecord(LocalDate.now(), stats.getPlayedTimes(), stats.getZombiesKilled(), stats.getDamageDealt(), state.getPlayer().getDamageTaken(), victory);

        List<MatchRecord> previousBest = matchDAO.bestSurvivalTimes(1);
        long previousBestMs = previousBest.isEmpty() ? -1 : previousBest.get(0).getSurvivalTimeMs();

        this.matchTimeMs = record.getSurvivalTimeMs();
        this.newRecord = matchTimeMs > previousBestMs;
        this.bestTimeMs = Math.max(matchTimeMs, previousBestMs);

        matchDAO.save(record);
    }

    public long getMatchTimeMs() {
        return matchTimeMs;
    }
    public long getBestTimeMs() {
        return bestTimeMs;
    }
    public boolean isNewRecord() {
        return newRecord;
    }

    public GameState getState() {
        return state;
    }
    public GameStatus getStatus() {
        return status;
    }
}

package model;

import java.io.InputStream;
import java.util.Scanner;

public class MapLoader {

    public static boolean[][] loadCollisionMatrix(String filePath, int cols, int rows) {
        boolean[][] matrix = new boolean[cols][rows];
        try (InputStream is = MapLoader.class.getResourceAsStream(filePath)) {
            if (is == null) {
                System.err.println("No se encontró el mapa en: " + filePath);
                return matrix;
            }
            Scanner scanner = new Scanner(is, "UTF-8").useDelimiter("\\A");
            String json = scanner.hasNext() ? scanner.next() : "";
            scanner.close();

            int startData = json.indexOf("\"data\":[");
            if (startData == -1) {
                System.err.println("ERROR MapLoader: No se encontró ningún array 'data'");
                return matrix;
            }
            int startBracket = json.indexOf("[", startData);
            int endBracket = json.indexOf("]", startBracket);

            if (startBracket == -1 || endBracket == -1) {
                System.err.println("ERROR MapLoader: No se encontraron los corchetes de data");
                return matrix;
            }

            String dataString = json.substring(startBracket + 1, endBracket);
            String[] values = dataString.split(",");

            System.out.println("MapLoader: Total de tiles leídos en la capa = " + values.length);

            int solidCount = 0;
            int col = 0;
            int row = 0;

            for (String val : values) {
                String trimmed = val.trim();
                if (!trimmed.isEmpty()) {
                    long id = Long.parseLong(trimmed);
                    boolean isSolid = (id > 0);

                    if (col < cols && row < rows) {
                        matrix[col][row] = isSolid;
                        if (isSolid) solidCount++;
                    }

                    col++;
                    if (col >= cols) {
                        col = 0;
                        row++;
                    }
                }
            }

            System.out.println("MapLoader: Carga completada. Obstáculos sólidos detectados: " + solidCount);
        } catch (Exception e) {
            System.err.println("Error leyendo el mapa: " + e.getMessage());
            e.printStackTrace();
        }

        return matrix;
    }
}

package model;

public class MeleeWeapon extends Weapon {

    private final double range = 100.0; // Píxeles de alcance del cuchillo
    private final double arcAngle = Math.toRadians(90); // 90 grados de arco de ataque

    public MeleeWeapon(int damage, long cooldown) {
        super(damage);
        this.cooldown = cooldown;
    }

    @Override
    public boolean attack(Player player, GameState state) {
        long currentTime = System.currentTimeMillis();

        if (currentTime - this.lastShoot >= this.cooldown) {
            this.lastShoot = currentTime;

            double totalDamage = this.getDamage() + player.getDamageBonus();
            boolean hitSomething = false;

            for (Enemy zombie : state.getEnemies()) {
                if (!zombie.isAlive()) continue;

                double playerCenterX = player.getX() + (player.getWidth() / 2.0);
                double playerCenterY = player.getY() + (player.getHeight() / 2.0);

                double zombieCenterX = zombie.getX() + (zombie.getWidth() / 2.0);
                double zombieCenterY = zombie.getY() + (zombie.getHeight() / 2.0);

                double dx = zombieCenterX - playerCenterX;
                double dy = zombieCenterY - playerCenterY;

                double distance = Math.hypot(dx, dy);

                if (distance <= range) {
                    double angleToZombie = Math.atan2(dy, dx); // Calculo el ángulo hacia el zombie

                    double angleDiff = angleToZombie - player.getAngle();
                    angleDiff = Math.atan2(Math.sin(angleDiff), Math.cos(angleDiff));

                    if (Math.abs(angleDiff) <= this.arcAngle / 2.0) {
                        int healthBefore = zombie.getHealth();
                        zombie.takeDamage((int) totalDamage);
                        state.getMatchStats().addDamageDealt(healthBefore - zombie.getHealth());
                        hitSomething = true;
                    }
                }
            }
            return hitSomething;
        }
        return false;
    }

    @Override
    public boolean isAutomatic() {
        return false;
    }
}

package model;

import java.util.ArrayList;
import java.util.Iterator;
import java.util.List;
import java.util.function.UnaryOperator;

public class Player extends Character {

    private boolean moveUp;
    private boolean moveDown;
    private boolean moveLeft;
    private boolean moveRight;
    private boolean isAttacking;

    // [POWERUP] estado que modifican los power ups
    private final List<Effect> effects = new ArrayList<>();
    private PlayerStats stats = new BaseStats(); // cadena de decoradores (escudo, daño, munición)
    private AmmoHolder ammoHolder; // arma actual (puede ser null hasta que exista Weapon)

    private double angle; // Ángulo en radianes hacia el cursor

    private final MeleeWeapon knife;
    private FireArm primaryWeapon; // Arma de fuego equipada
    private Weapon activeWeapon; // Puntero al arma que tiene en la mano actualmente

    // --- VARIABLES DE INMUNIDAD (I-FRAMES) ---
    private long lastDamageTime = 0;
    private final int invulnerabilityCooldown = 1000; // 1 segundo de inmunidad al chocar

    private int damageTaken = 0;

    public Player(double x, double y) {
        super(x, y, 64, 64, 8 , 100, 100);

        this.knife = new MeleeWeapon(50, 1000);
        this.activeWeapon = this.knife;
    }

    public void setMoveUp(boolean moveUp) {
        this.moveUp = moveUp;
    }

    public void setMoveDown(boolean moveDown) {
        this.moveDown = moveDown;
    }

    public void setMoveLeft(boolean moveLeft) {
        this.moveLeft = moveLeft;
    }

    public void setMoveRight(boolean moveRight) {
        this.moveRight = moveRight;
    }

    public void setAttacking(boolean attacking) {
        this.isAttacking = attacking;
    }

    public boolean isMoving() {
        return moveUp || moveDown || moveRight || moveLeft;
    }

    public double getAngle() {
        return angle;
    }

    public FireArm getPrimaryWeapon() {
        return primaryWeapon;
    }

    public Weapon getActiveWeapon() {
        return activeWeapon;
    }

    public void update(GameState state) {
        updateEffects(); // [POWERUP] vence los efectos temporales
        stats = stats.prune(); // [POWERUP] descarta los decoradores vencidos

        if (this.isAttacking && this.activeWeapon != null && this.activeWeapon.isAutomatic()) {
            this.attack(state);
        }

        double dx = 0;
        double dy = 0;
        if (moveUp) dy -= 1;
        if (moveDown) dy += 1;
        if (moveLeft) dx -= 1;
        if (moveRight) dx += 1;

        move(dx, dy, state);

        // Bordes del mapa calculados 
        if (x < 0) x = 0;
        if (x + width > state.getMapWidth()) x = state.getMapWidth() - width;
        if (y < 0) y = 0;
        if (y + height > state.getMapHeight()) y = state.getMapHeight() - height;
    }

    public void aimAt(double mouseX, double mouseY) {
        //Centro real del personaje
        double centerX = this.getX() + (this.getWidth() / 2.0);
        double centerY = this.getY() + (this.getHeight() / 2.0);

        double deltaX = mouseX - centerX;
        double deltaY = mouseY - centerY;

        this.angle = Math.atan2(deltaY, deltaX);
    }

    public void swapWeapon() {
        if (this.activeWeapon == this.knife && this.primaryWeapon != null) {
            this.activeWeapon = this.primaryWeapon;
            this.setAmmoHolder(this.primaryWeapon);
        } else {
            this.activeWeapon = this.knife;
            this.setAmmoHolder(null);
        }
    }

    public void equipWeapon(FireArm weapon) {
        this.primaryWeapon = weapon;
        this.activeWeapon = this.primaryWeapon;
        this.setAmmoHolder(weapon);
    }

    public void attack(GameState state) {
        if (this.activeWeapon != null) {
            this.activeWeapon.attack(this, state);
        }
    }

    public void addEffect(Effect effect) {
        effect.apply(this);
        effects.add(effect);
    }

    private void updateEffects() {
        Iterator<Effect> it = effects.iterator();
        while (it.hasNext()) {
            Effect e = it.next();
            e.update(this);
            if (e.isExpired()) {
                e.remove(this);
                it.remove();
            }
        }
    }

    /** Envuelve las stats actuales con un nuevo decorador, ej: player.decorateStats(s -> new ShieldDecorator(s, 50)). */
    public void decorateStats(UnaryOperator<PlayerStats> decorator) {
        stats = decorator.apply(stats);
        stats.onApply(this);
    }

    public double getDamageBonus() {
        return stats.getDamageBonus();
    }

    public void heal(int amount) {
        setHealth(Math.min(getHealth() + amount, getMaxHealth()));
    }

    public int getShield() {
        return stats.getShield();
    }

    @Override
    public void takeDamage(int amount) {
        long currentTime = System.currentTimeMillis();
        
        // Evaluamos el cooldown de choque antes de procesar el daño
        if (currentTime - lastDamageTime >= invulnerabilityCooldown) {
            // Los decoradores de escudo absorben primero; a la vida llega lo que sobra
            int reachesHealth = stats.absorbDamage(amount);
            int absorbedByShield = amount - reachesHealth;

            int healthBefore = getHealth();
            super.takeDamage(reachesHealth);

            damageTaken += absorbedByShield + (healthBefore - getHealth());

            // Reiniciamos el temporizador de inmunidad
            this.lastDamageTime = currentTime;
        }
    }

    public void setAmmoHolder(AmmoHolder ammoHolder) {
        this.ammoHolder = ammoHolder;
    }

    public void addAmmo(int amount) {
        if (ammoHolder != null) ammoHolder.addAmmo(amount);
    }

    public void refillAmmo() {
        if (this.ammoHolder != null) {
            this.ammoHolder.refillAmmo();
        }
    }

    public int getDamageTaken() {
        return damageTaken;
    }
}

package model;

import java.util.List;
import java.util.Random;

/**
 * Genera power ups repartidos por TODO el mapa, solo en lugares a los que el jugador puede llegar caminando
 * (respeta la matriz de colisiones: nada sobre obstáculos ni dentro de zonas cerradas).
 * Al crearse deja una cantidad inicial y después repone de a uno cada cierto tiempo.
 * Mismo estilo que EnemySpawner.
 */
public class PowerUpSpawner {

    private static final int TYPES = 5;                         // Heal, Shield, DamageBoost, Ammo, MaxAmmo
    private static final int MAX_ATTEMPTS = 200;                // intentos para encontrar un lugar válido
    private static final double MIN_DIST_TO_PLAYER = 250;       // no aparece encima del jugador
    private static final double MIN_DIST_BETWEEN = 150;         // separación mínima entre power ups

    private final GameState state;
    private final ReachableArea reachableArea; // zonas a las que el jugador puede llegar
    private final List<PowerUp> powerUps;
    private final long spawnInterval; // ms entre cada reposición
    private final int maxOnMap;       // tope de power ups simultáneos
    private long lastSpawnTime;
    private final Random random = new Random();

    public PowerUpSpawner(GameState state, long spawnInterval, int maxOnMap, int initialCount) {
        this.state = state;
        // La matriz de colisiones y la posición inicial del jugador ya tienen que estar cargadas acá
        this.reachableArea = new ReachableArea(state);
        if (!reachableArea.isUsable()) {
            System.err.println("PowerUpSpawner: no se pudo calcular la zona alcanzable; se usa solo la matriz de colisiones.");
        }
        this.powerUps = state.getPowerUps();
        this.spawnInterval = spawnInterval;
        this.maxOnMap = maxOnMap;
        this.lastSpawnTime = System.currentTimeMillis();
        populate(Math.min(initialCount, maxOnMap));
    }

    /** Reparte la cantidad inicial recorriendo los tipos en orden, para que haya variedad. */
    private void populate(int count) {
        int firstType = random.nextInt(TYPES);
        for (int i = 0; i < count; i++) {
            trySpawn((firstType + i) % TYPES);
        }
    }

    public void update() {
        long now = System.currentTimeMillis();
        if (now - lastSpawnTime < spawnInterval || powerUps.size() >= maxOnMap) return;
        lastSpawnTime = now;
        trySpawn(random.nextInt(TYPES));
    }

    private boolean trySpawn(int type) {
        int mapWidth = state.getMapWidth();
        int mapHeight = state.getMapHeight();
        if (mapWidth <= PowerUp.SIZE || mapHeight <= PowerUp.SIZE) return false;

        for (int attempt = 0; attempt < MAX_ATTEMPTS; attempt++) {
            double x = random.nextInt(mapWidth - PowerUp.SIZE);
            double y = random.nextInt(mapHeight - PowerUp.SIZE);
            if (isValidSpot(x, y)) {
                powerUps.add(create(type, x, y));
                return true;
            }
        }
        return false; // no encontró lugar: lo intenta en la próxima reposición
    }

    private boolean isValidSpot(double x, double y) {
        if (isOnBlockedTile(x, y)) return false;

        double centerX = x + PowerUp.SIZE / 2.0;
        double centerY = y + PowerUp.SIZE / 2.0;

        Player player = state.getPlayer();
        if (reachableArea.isUsable() && !reachableArea.canPickUpAt(centerX, centerY, player)) return false;

        double playerX = player.getX() + player.getWidth() / 2.0;
        double playerY = player.getY() + player.getHeight() / 2.0;
        if (Math.hypot(centerX - playerX, centerY - playerY) < MIN_DIST_TO_PLAYER) return false;

        for (PowerUp other : powerUps) {
            double otherX = other.getX() + PowerUp.SIZE / 2.0;
            double otherY = other.getY() + PowerUp.SIZE / 2.0;
            if (Math.hypot(centerX - otherX, centerY - otherY) < MIN_DIST_BETWEEN) return false;
        }
        return true;
    }

    /** true si el power up toca algún tile bloqueado del mapa o se sale de él. */
    private boolean isOnBlockedTile(double x, double y) {
        boolean[][] map = state.getCollisionMatrix();
        if (map == null) return false;

        int tile = GameState.TILE_SIZE;
        int fromX = (int) x / tile;
        int toX = (int) (x + PowerUp.SIZE - 1) / tile;
        int fromY = (int) y / tile;
        int toY = (int) (y + PowerUp.SIZE - 1) / tile;

        for (int tx = fromX; tx <= toX; tx++) {
            for (int ty = fromY; ty <= toY; ty++) {
                if (tx < 0 || tx >= map.length || ty < 0 || ty >= map[0].length) return true;
                if (map[tx][ty]) return true;
            }
        }
        return false;
    }

    private PowerUp create(int type, double x, double y) {
        return switch (type) {
            case 0 -> new Heal(x, y, 25);
            case 1 -> new Shield(x, y, 50);
            case 2 -> new DamageBoost(x, y, 10000);
            case 3 -> new Ammo(x, y, 30);
            default -> new MaxAmmo(x, y);
        };
    }
}

package model;

import java.util.ArrayDeque;

/**
 * Calcula a qué zonas del mapa puede llegar realmente el jugador desde donde empieza.
 * Usa la MISMA regla de colisión con la que el jugador se mueve (Character.isCollidingWithMap),
 * así que si acá figura como alcanzable, el jugador puede llegar caminando.
 */
public class ReachableArea {

    private static final int STEP = 16;          // resolución de la grilla (px) // Thiago: cambio a 16 por motivos de rendimiento
    private static final int PICKUP_REACH = 32; // cuánto puede estar el centro del jugador del centro del ítem
    private static final int MIN_CELLS = 100;   // si da menos, el cálculo no es confiable

    private final boolean[][] reachable; // [col][fila] -> posición (esq. sup. izq.) del jugador
    private final int cols;
    private final int rows;
    private final int reachableCells;

    public ReachableArea(GameState state) {
        Player player = state.getPlayer();
        cols = (state.getMapWidth() - player.getWidth()) / STEP + 1;
        rows = (state.getMapHeight() - player.getHeight()) / STEP + 1;
        reachable = new boolean[Math.max(cols, 0)][Math.max(rows, 0)];

        if (cols <= 0 || rows <= 0) {
            reachableCells = 0;
            return;
        }

        int startCol = clamp(Math.round((float) player.getX() / STEP), 0, cols - 1);
        int startRow = clamp(Math.round((float) player.getY() / STEP), 0, rows - 1);

        ArrayDeque<int[]> queue = new ArrayDeque<>();
        reachable[startCol][startRow] = true;
        queue.add(new int[]{startCol, startRow});

        int count = 1;
        int[][] directions = {{1, 0}, {-1, 0}, {0, 1}, {0, -1}};

        while (!queue.isEmpty()) {
            int[] cell = queue.poll();
            for (int[] d : directions) {
                int c = cell[0] + d[0];
                int r = cell[1] + d[1];
                if (c < 0 || r < 0 || c >= cols || r >= rows || reachable[c][r]) continue;
                if (player.isCollidingWithMap(c * STEP, r * STEP, state)) continue;

                reachable[c][r] = true;
                queue.add(new int[]{c, r});
                count++;
            }
        }
        reachableCells = count;
    }

    /** false si el cálculo no sirvió (ej: el jugador empieza encerrado); en ese caso no hay que usarlo. */
    public boolean isUsable() {
        return reachableCells >= MIN_CELLS;
    }

    /** true si el jugador puede llegar caminando lo bastante cerca como para agarrar un ítem centrado en (itemX, itemY). */
    public boolean canPickUpAt(double itemCenterX, double itemCenterY, Player player) {
        double playerLeft = itemCenterX - player.getWidth() / 2.0;
        double playerTop = itemCenterY - player.getHeight() / 2.0;

        int fromCol = (int) Math.floor((playerLeft - PICKUP_REACH) / STEP);
        int toCol = (int) Math.ceil((playerLeft + PICKUP_REACH) / STEP);
        int fromRow = (int) Math.floor((playerTop - PICKUP_REACH) / STEP);
        int toRow = (int) Math.ceil((playerTop + PICKUP_REACH) / STEP);

        for (int c = Math.max(fromCol, 0); c <= Math.min(toCol, cols - 1); c++) {
            for (int r = Math.max(fromRow, 0); r <= Math.min(toRow, rows - 1); r++) {
                if (reachable[c][r]) return true;
            }
        }
        return false;
    }

    private static int clamp(int value, int min, int max) {
        return Math.max(min, Math.min(value, max));
    }
}

/** Agrega puntos de escudo: absorbe daño antes de que llegue a la vida. Se descarta al agotarse. */
public class ShieldDecorator extends StatsDecorator {

    public static final int MAX_SHIELD = 100; // tope del escudo total del jugador

    private int points;

    public ShieldDecorator(PlayerStats wrapped, int points) {
        super(wrapped);
        // Respeta el tope: lo que ya hay en la cadena cuenta para el máximo
        int room = Math.max(0, MAX_SHIELD - wrapped.getShield());
        this.points = Math.max(0, Math.min(points, room));
    }

    @Override
    public int getShield() {
        return points + wrapped.getShield();
    }

    @Override
    public int absorbDamage(int amount) {
        int absorbed = Math.min(points, amount);
        points -= absorbed;
        // lo que no absorbí pasa al siguiente decorador de la cadena
        return wrapped.absorbDamage(amount - absorbed);
    }

    @Override
    public boolean isExpired() {
        return points <= 0;
    }
}

package model;
 
import java.util.Random;
 
public class Shotgun extends FireArm {
 
    private final Random random = new Random();
 
    public Shotgun(int damage, int cooldown, int actualMunition, int maxMunition) {
        super(damage, cooldown, actualMunition, maxMunition);
    }
 
    @Override
    public boolean attack(Player player, GameState state) {
        long currentTime = System.currentTimeMillis();
 
        if (this.actualMunition > 0 && currentTime - this.lastShoot >= this.cooldown) {
 
            this.actualMunition--;
            this.lastShoot = currentTime;
 
            double totalDamage = this.getDamage() + player.getDamageBonus();

            for (int i = 0; i < 4; i++) {
                // random.nextDouble() da un valor entre 0.0 y 1.0.
                // Al restar 0.5 queda entre -0.5 y 0.5.
                // Multiplicado por 0.6 genera una dispersión máxima de +/- 0.3 radianes.
                double offset = (random.nextDouble() - 0.5) * 0.6;
                double spreadAngle = player.getAngle() + offset;
 
                double dirX = Math.cos(spreadAngle);
                double dirY = Math.sin(spreadAngle);
 
                double muzzleX = getMuzzleX(player);
                double muzzleY = getMuzzleY(player);
 
                Bullet pellet = new Bullet(muzzleX, muzzleY, 6, 6, dirX, dirY, totalDamage, 30, BulletKind.PELLET);
                state.addPlayerBullet(pellet);
            }
            return true;
        }
        return false;
    }

    @Override 
    public boolean isAutomatic() {
        return false;
    }
}

/**
 * DECORADOR base: implementa PlayerStats y envuelve a otro PlayerStats.
 * Por defecto delega todo en el objeto envuelto; cada decorador concreto
 * redefine solo lo que le interesa.
 */
public abstract class StatsDecorator implements PlayerStats {

        this.wrapped = wrapped;
    }

    public double getDamageBonus() {
        return wrapped.getDamageBonus();
    }

    @Override
    public int getShield() {
        return wrapped.getShield();
    }

    @Override
    public int absorbDamage(int amount) {

    }

    @Override
    public PlayerStats prune() {
        wrapped = wrapped.prune();
        return isExpired() ? wrapped : this;
    }

package persistence;

import model.MatchDAO;
import model.MatchRecord;

import java.sql.*;
import java.time.LocalDate;
import java.util.ArrayList;
import java.util.List;

public class MatchDAOSQLite implements MatchDAO {

    private final Connection connection;

    public MatchDAOSQLite(Connection connection) throws SQLException {
        if (connection == null) {
            throw new IllegalArgumentException("La conexión es obligatoria");
        }
        this.connection = connection;

        try (Statement statement = connection.createStatement()) {
            statement.execute("CREATE TABLE IF NOT EXISTS matches ("
                    + "id INTEGER PRIMARY KEY AUTOINCREMENT, "
                    + "match_date TEXT NOT NULL, "
                    + "survival_time_ms INTEGER NOT NULL, "
                    + "zombies_killed INTEGER NOT NULL, "
                    + "damage_dealt INTEGER NOT NULL, "
                    + "damage_taken INTEGER NOT NULL, "
                    + "victory INTEGER NOT NULL)");
        }
    }

    @Override
    public boolean save(MatchRecord match) {
        if (match == null) {
            return false;
        }

        String sql = "INSERT INTO matches (match_date, survival_time_ms, zombies_killed, damage_dealt, damage_taken, victory)"
                + "VALUES (?, ?, ?, ?, ?)";

        try (PreparedStatement statement = this.connection.prepareStatement(sql)) {
            statement.setString(1, match.getDate().toString());
            statement.setLong(2, match.getSurvivalTimeMs());
            statement.setInt(3, match.getZombiesKilled());
            statement.setInt(4, match.getDamageDealt());
            statement.setInt(5, match.getDamageTaken());
            statement.setInt(6, match.isVictory() ? 1 : 0);
            return statement.executeUpdate() == 1;
        } catch (SQLException e) {
            return false;
        }
    }

    @Override
    public List<MatchRecord> bestSurvivalTimes(int limit) {
        List<MatchRecord> result = new ArrayList<>();

        if (limit <= 0) {
            return result;
        }

        String sql = "SELECT match_date, survival_time_ms, zombies_killed, damage_dealt,"
                + "damage_taken, victory FROM matches ORDER BY survival_time_ms DESC LIMIT ?";

        try (PreparedStatement statement = this.connection.prepareStatement(sql)) {
            statement.setInt(1, limit);

            try (ResultSet resultSet = statement.executeQuery()) {
                while (resultSet.next()) {
                    result.add(new MatchRecord(
                            LocalDate.parse(resultSet.getString("match_date")),
                            resultSet.getLong("survival_time_ms"),
                            resultSet.getInt("zombies_killed"),
                            resultSet.getInt("damage_dealt"),
                            resultSet.getInt("damage_taken"),
                            resultSet.getInt("victory") == 1));
                }
            }

        } catch (SQLException e) {
            return result;
        }
        return result;
    }
}

package view;

import java.awt.image.BufferedImage;

public class Animation {
    private final BufferedImage[] frames;
    private int currentFrame;
    private long lastTime;
    private final int delay; // Milisegundos entre cada fotograma

    public Animation(int delay, BufferedImage... frames) {
        this.delay = delay;
        this.frames = frames;
        this.lastTime = System.currentTimeMillis();
    }

    public void update() {
        long timer = System.currentTimeMillis() - this.lastTime;
        if (timer > this.delay) {
            this.currentFrame ++;
            if (this.currentFrame >= this.frames.length) {
                this.currentFrame = 0; // Vuelve al inicio para hacer un bucle
            }
            lastTime = System.currentTimeMillis();
        }
    }

    public BufferedImage getCurrentFrame() {
        return this.frames[this.currentFrame];
    }
}

package view;

import java.awt.BasicStroke;

/**
  Efectos visuales de disparo: estelas (tracers) de las balas y destello en la boca del arma.
  Solo dibuja; no modifica el modelo. Hay que llamarlo con el Graphics2D en coordenadas de MUNDO.
 */

public class BulletEffects {
 
    private static final long FLASH_MS = 70;
    private static final long SHOTGUN_FLASH_MS = 100;
 
    // ESTELAS
    public void drawTracer(Graphics2D g2, double x, double y, int w, int h, double dirX, double dirY, double speed, int age, String kind) {
        Stroke oldStroke = g2.getStroke();
        Paint oldPaint = g2.getPaint();

        boolean pellet = kind.equals("PELLET");
 
        double maxLength = pellet ? 18 : 36;
        // La estela crece desde 0 los primeros ticks: parece que "sale" del arma
        double length = Math.min(maxLength, age * speed + 6);
 
        double headX = x + w / 2.0;
        double headY = y + h / 2.0;
        double tailX = headX - dirX * length;
        double tailY = headY - dirY * length;
 
        Color glow = pellet ? new Color(255, 110, 20) : new Color(255, 185, 60);
        Line2D line = new Line2D.Double(tailX, tailY, headX, headY);
 
        // Halo exterior
        g2.setPaint(new GradientPaint((float) tailX, (float) tailY, alpha(glow, 0), (float) headX, (float) headY, alpha(glow, 90)));
        g2.setStroke(new BasicStroke(pellet ? 7f : 9f, BasicStroke.CAP_ROUND, BasicStroke.JOIN_ROUND));
        g2.draw(line);
 
        // Cuerpo
        g2.setPaint(new GradientPaint((float) tailX, (float) tailY, alpha(glow, 0), (float) headX, (float) headY, alpha(glow, 210)));
        g2.setStroke(new BasicStroke(4f, BasicStroke.CAP_ROUND, BasicStroke.JOIN_ROUND));
        g2.draw(line);
 
        // Núcleo blanco-amarillento
        Color core = new Color(255, 250, 210);
        g2.setPaint(new GradientPaint((float) tailX, (float) tailY, alpha(core, 0), (float) headX, (float) headY, alpha(core, 255)));
        g2.setStroke(new BasicStroke(2f, BasicStroke.CAP_ROUND, BasicStroke.JOIN_ROUND));
        g2.draw(line);
 
        // Punta brillante
        g2.setPaint(alpha(core, 235));
        g2.fill(new Ellipse2D.Double(headX - 2.5, headY - 2.5, 5, 5));

        g2.setStroke(oldStroke);
        g2.setPaint(oldPaint);
    }
 
    // DESTELLO DE BOCA DE FUEGO
    public void drawMuzzleFlash(Graphics2D g2, double x, double y, int w, int h, double angle, String weaponType, long lastShoot, double muzzleForward, double muzzleSide) {

        boolean shotgun = weaponType.equals("Shotgun");
        long duration = shotgun ? SHOTGUN_FLASH_MS : FLASH_MS;
        long elapsed = System.currentTimeMillis() - lastShoot;
        if (elapsed < 0 || elapsed > duration) return;
 
        double life = 1.0 - (double) elapsed / duration; // 1 al disparar -> 0 al apagarse
 
        double centerX = x + w / 2.0;
        double centerY = y + h / 2.0;
 
        AffineTransform old = g2.getTransform();
        Paint oldPaint = g2.getPaint();
        Stroke oldStroke = g2.getStroke();
 
        // A partir de acá, el eje X apunta hacia donde mira el jugador y (0,0) es la boca del arma
        g2.translate(centerX, centerY);
        g2.rotate(angle);
        g2.translate(muzzleForward, muzzleSide);
 
        // Misma semilla durante todo el destello: la forma se mantiene y solo se desvanece
        Random rnd = new Random(lastShoot);
 
        drawFlashGlow(g2, life, shotgun);
        drawFlashSpikes(g2, rnd, life, shotgun);
        drawFlashSparks(g2, rnd, life, shotgun);
 
        g2.setTransform(old);
        g2.setPaint(oldPaint);
        g2.setStroke(oldStroke);
    }
 
    /** Resplandor redondo + luz tenue que "ilumina" la zona oscura alrededor. */
    private void drawFlashGlow(Graphics2D g2, double life, boolean shotgun) {
        Point2D center = new Point2D.Double(0, 0);
        float[] fractions = {0f, 0.35f, 1f};
 
        // Iluminación ambiente
        double ambient = shotgun ? 190 : 140;
        g2.setPaint(new RadialGradientPaint(center, (float) ambient, fractions, new Color[]{
                new Color(255, 170, 70, (int) (70 * life)),
                new Color(255, 120, 30, (int) (35 * life)),
                new Color(255, 100, 0, 0)}));
        g2.fill(new Ellipse2D.Double(-ambient, -ambient, ambient * 2, ambient * 2));
 
        // Núcleo brillante
        double radius = (shotgun ? 70 : 50) * (0.6 + 0.4 * life);
        g2.setPaint(new RadialGradientPaint(center, (float) radius, fractions, new Color[]{
                new Color(255, 245, 190, (int) (230 * life)),
                new Color(255, 150, 40, (int) (150 * life)),
                new Color(255, 100, 0, 0)}));
        g2.fill(new Ellipse2D.Double(-radius, -radius, radius * 2, radius * 2));
    }
 
    /** Picos de fuego en abanico hacia adelante. */
    private void drawFlashSpikes(Graphics2D g2, Random rnd, double life, boolean shotgun) {
        int count = shotgun ? 9 : 6;
        double spread = shotgun ? 0.75 : 0.5;     // apertura del abanico (rad)
        double baseLength = shotgun ? 70 : 48;
 
        Path2D outer = new Path2D.Double();
        Path2D inner = new Path2D.Double();
 
        for (int i = 0; i < count; i++) {
            // El primer pico va derecho (el más largo); el resto se reparte en abanico
            double angle = (i == 0) ? 0 : (rnd.nextDouble() * 2 - 1) * spread;
            double length = baseLength * (i == 0 ? 1.2 : 0.45 + rnd.nextDouble() * 0.6) * (0.5 + 0.5 * life);
            double halfWidth = (i == 0 ? 7 : 4) * (0.6 + 0.4 * life);
 
            addSpike(outer, angle, length, halfWidth);
            addSpike(inner, angle, length * 0.55, halfWidth * 0.5);
        }
 
        g2.setPaint(new Color(255, 160, 40, (int) (230 * life)));
        g2.fill(outer);
        g2.setPaint(new Color(255, 250, 215, (int) (255 * life)));
        g2.fill(inner);
    }
 
    /** Triángulo fino desde la boca del arma hacia el ángulo indicado. */
    private void addSpike(Path2D path, double angle, double length, double halfWidth) {
        double dx = Math.cos(angle);
        double dy = Math.sin(angle);
        double px = -dy * halfWidth; // perpendicular
        double py = dx * halfWidth;
 
        path.moveTo(px, py);
        path.lineTo(dx * length, dy * length);
        path.lineTo(-px, -py);
        path.closePath();
    }
 
    /** Chispas chicas que salen disparadas y se alejan mientras el destello se apaga. */
    private void drawFlashSparks(Graphics2D g2, Random rnd, double life, boolean shotgun) {
        int count = shotgun ? 9 : 5;
        double progress = 1.0 - life; // 0 -> 1
 
        g2.setStroke(new BasicStroke(1.6f, BasicStroke.CAP_ROUND, BasicStroke.JOIN_ROUND));
 
        for (int i = 0; i < count; i++) {
            double angle = (rnd.nextDouble() * 2 - 1) * 0.9;
            double maxDist = 30 + rnd.nextDouble() * (shotgun ? 70 : 45);
            double start = 6 + progress * maxDist;
            double end = start + 5 + rnd.nextDouble() * 7;
 
            g2.setPaint(new Color(255, 215, 120, (int) (255 * life)));
            g2.draw(new Line2D.Double(
                    Math.cos(angle) * start, Math.sin(angle) * start,
                    Math.cos(angle) * end, Math.sin(angle) * end));
        }
    }
 
    private static Color alpha(Color c, int a) {
        return new Color(c.getRed(), c.getGreen(), c.getBlue(), Math.max(0, Math.min(255, a)));
    }
}

    // Dibuja al jugador
    public void drawPlayer(double x, double y, int w, int h, double angle, boolean moving, String weaponType) {
        AffineTransform transformOriginal = g2.getTransform();

        double centerX = x + (w / 2.0);
        double centerY = y + (h / 2.0);

        g2.translate(centerX, centerY);
        g2.rotate(angle);

        if (moving && playerFeetAnimation != null && playerFeetAnimation.getCurrentFrame() != null) {
            BufferedImage feetFrame = playerFeetAnimation.getCurrentFrame();
            AffineTransform transformFeet = g2.getTransform();
            g2.scale(0.6, 0.6);
            g2.drawImage(feetFrame, -w / 2, -h / 2, w, h, null);
            g2.setTransform(transformFeet);
        }

        Animation currentAnim = playerAnimations.get(weaponType);

        if (currentAnim != null) {
            currentAnim.update();
            if (currentAnim.getCurrentFrame() != null) {
                BufferedImage bodyFrame = currentAnim.getCurrentFrame();
                g2.drawImage(bodyFrame, -w / 2, -h / 2, w, h, null);
            }
        } else {
            g2.setColor(Color.RED);
            g2.fillOval(-w / 2, -h / 2, w, h);
        }

        g2.setTransform(transformOriginal);
    }

    // NOTA: Se agregó el parámetro 'boolean isAttacking'
    public void drawZombie(double x, double y, int w, int h, double angle, int health, String type, boolean isAttacking) {
        double spriteAngle = angle + SPRITE_ROTATION.getOrDefault(type, 0.0);

        AffineTransform transformOriginal = g2.getTransform();
        double centerX = x + (w / 2.0);
        double centerY = y + (h / 2.0);

        g2.translate(centerX, centerY);
        g2.rotate(spriteAngle);

        Animation attackAnim = zombieAttackAnimations.get(type);
        Animation walkAnim = zombieAnimations.get(type);
        BufferedImage idle = zombieIdleSprites.get(type);

        BufferedImage frameToDraw = null;

        // Lógica de prioridad: Ataque -> Caminar -> Reposo -> Círculo rojo de error
        if (isAttacking && attackAnim != null && attackAnim.getCurrentFrame() != null) {
            frameToDraw = attackAnim.getCurrentFrame();
        } else if (walkAnim != null && walkAnim.getCurrentFrame() != null) {
            frameToDraw = walkAnim.getCurrentFrame();
        } else if (idle != null) {
            frameToDraw = idle;
        }

        if (frameToDraw != null) {
            g2.drawImage(frameToDraw, -w / 2, -h / 2, w, h, null);
        } else {
            g2.setColor(Color.RED);
            g2.fillOval(-w / 2, -h / 2, w, h);
        }

        g2.setTransform(transformOriginal);
        g2.setColor(Color.WHITE);
        g2.setFont(new Font("Monospaced", Font.BOLD, 20));
        g2.drawString("HP: " + health, (int) x, (int) y);
    }

    // Oscuridad
    public void drawDarkness(double playerX, double playerY, int playerW, int playerH, double playerAngle, int mapWidth, int mapHeight) {
        double centerX = playerX + (playerW / 2.0);
        double centerY = playerY + (playerH / 2.0);

        Area darkness = new Area(new Rectangle2D.Double(0, 0, mapWidth, mapHeight));

        double lightRadius = 700.0; // Define distancia
        double fovAngle = 85.0; // Define ángulo

        double angleDegrees = -Math.toDegrees(playerAngle);

        Arc2D.Double lightCone = new Arc2D.Double(
                centerX - lightRadius, centerY - lightRadius,
                lightRadius * 2, lightRadius * 2,
                angleDegrees - (fovAngle / 2.0), fovAngle,
                Arc2D.PIE
        );

        double personalRadius = 90.0; // Círculo personal donde se va a ver
        Ellipse2D.Double personalCircle = new Ellipse2D.Double(
                centerX - personalRadius, centerY - personalRadius,
                personalRadius * 2, personalRadius * 2
        );

        darkness.subtract(new Area(lightCone)); // Substrae el cono de iluminación
        darkness.subtract(new Area(personalCircle)); // Substrae el círculo personal

        g2.setColor(new Color(0, 0, 0, 220)); // 245 = opacidad
        g2.fill(darkness);

        g2.setStroke(new BasicStroke(1.0f)); // Restaurar el grosor normal
    }

    // Pantalla de intro del jefe final
    public void drawBossIntro(int step, double progress) {
        // el texto aparece y se desvanece de a poco
        double fade = Math.max(0.0, Math.min(1.0, Math.min(progress / 0.25, (1.0 - progress) / 0.25)));
        int alpha = (int) (255 * fade);

        // fondo oscuro
        g2.setColor(new Color(0, 0, 0, 215));
        g2.fillRect(0, 0, getWidth(), getHeight());

        String text;
        Color color;
        int size;
        if (step == 0) {
            text = "Felicidades, sobreviviste a la primer oleada, buen trabajo";
            color = new Color(190, 190, 190);
            size = 32;
        } else {
            text = "Preparate para el Jefe";
            color = new Color(190, 0, 0);
            size = 54;
        }

        // si el texto no entra en pantalla, se achica la letra
        Font font = new Font("Serif", Font.BOLD | Font.ITALIC, size);
        FontMetrics fm = g2.getFontMetrics(font);
        while (fm.stringWidth(text) > getWidth() - 80 && size > 12) {
            size--;
            font = new Font("Serif", Font.BOLD | Font.ITALIC, size);
            fm = g2.getFontMetrics(font);
        }
        g2.setFont(font);

        int x = (getWidth() - fm.stringWidth(text)) / 2;
        int y = getHeight() / 2;

        // temblor leve solo en "Prepárate para el Jefe"
        if (step == 1) {
            x += (int) (Math.random() * 5) - 2;
            y += (int) (Math.random() * 5) - 2;
        }

        // sombra rojo oscuro y texto encima
        g2.setColor(new Color(70, 0, 0, alpha));
        g2.drawString(text, x + 3, y + 3);
        g2.setColor(new Color(color.getRed(), color.getGreen(), color.getBlue(), alpha));
        g2.drawString(text, x, y);
    }

    // Efectos para los power-ups
    private void drawFloating(Graphics2D g2, double x, double y, int w, int h, Color glowColor, Consumer<Graphics2D> content) {
        double t = System.currentTimeMillis() / 1000.0;
        double phase = x * 0.05;                               // desfase por objeto
        double bob = Math.sin(t * 3.0 + phase) * 4;                   // sube y baja 4 px
        double scale = 1.0 + Math.sin(t * 4.0 + phase) * 0.06;        // pulso de +-6%
        int glowAlpha = (int) (70 + Math.sin(t * 4.0 + phase) * 40);  // brillo 30..110

        AffineTransform old = g2.getTransform();
        g2.translate(x + w / 2.0, y + h / 2.0 + bob);

        g2.setColor(new Color(glowColor.getRed(), glowColor.getGreen(), glowColor.getBlue(), glowAlpha));
        g2.fillOval(-w / 2 - 6, -h / 2 - 6, w + 12, h + 12);

        g2.scale(scale, scale);
        content.accept(g2);       // cada uno dibuja lo suyo, centrado en (0, 0)

        g2.setTransform(old);
    }

package view;

import java.awt.*;
import java.awt.event.MouseListener;
import java.awt.event.MouseAdapter;
import java.awt.event.MouseEvent;

import javax.swing.JButton;

public class MenuButton extends JButton{

    private boolean hover;
    private static final Color NORMAL_BG_COLOR = new Color(15, 15, 15, 200);
    private static final Color HOVER_BG_COLOR = new Color(40, 10, 10, 230);
    private static final Color BORDER_COLOR = new Color(200, 0, 0);
    private static final Color NORMAL_TEXT_COLOR = new Color(200, 200, 200);
    private static final Color HOVER_TEXT_COLOR = Color.WHITE;

    public MenuButton(String text) {
        super(text);
        this.hover = false;

        setFont(new Font("Arial", Font.BOLD, 22));
        setForeground(NORMAL_TEXT_COLOR);

        setFocusPainted(false);
        setContentAreaFilled(false);
        setBorderPainted(false);
        setOpaque(false);
        setCursor(new Cursor(Cursor.CROSSHAIR_CURSOR));
        setPreferredSize(new Dimension(250, 50));
        setMaximumSize(new Dimension(250, 50));

        setAlignmentX(Component.CENTER_ALIGNMENT);

        addMouseListener((MouseListener) new MouseAdapter() {
            @Override 
            public void mouseEntered(MouseEvent e) {
                hover = true;
                setForeground(HOVER_TEXT_COLOR);
                repaint();
            }

            @Override 
            public void mouseExited(MouseEvent e) {
                hover = false;
                setForeground(NORMAL_TEXT_COLOR);
                repaint();
            }
        });
    }

    @Override 
    protected void paintComponent(Graphics g) {
        Graphics2D g2 = (Graphics2D) g.create();
        g2.setRenderingHint(RenderingHints.KEY_ANTIALIASING, RenderingHints.VALUE_ANTIALIAS_ON);

        int width = getWidth();
        int height = getHeight();
        // Fondo del botón
        g2.setColor(hover ? HOVER_BG_COLOR : NORMAL_BG_COLOR);
        g2.fillRect(2, 2, width - 4, height -4);
        // Borde exterior en rojo
        g2.setColor(BORDER_COLOR);
        // Si el mouse está encima, el borde se hace más grueso
        g2.setStroke(new BasicStroke(hover ? 3.0f : 1.0f));
        g2.drawRect(2, 2, width -4, height - 4);

        // Marco rojo tenue extra al hacer hover
        if (hover) {
            g2.setColor(new Color(255, 0, 0, 100));
            g2.setStroke(new BasicStroke(1.0f));
            g2.drawRect(6, 6, width - 12, height - 12);
        }

        g2.dispose();
        super.paintComponent(g);
    }
}

package view;

import javax.swing.*;
import java.awt.*;

public class MenuPanel extends JPanel {
    
    private Image backgroundImage;

    private MenuButton playButton;
    private MenuButton optionButton;
    private MenuButton exitButton;

    private java.util.List<Ceniza> particulas = new java.util.ArrayList<>();
    private javax.swing.Timer particleTimer;

    private class Ceniza {
        int x, y, size, speed;
        
        public Ceniza(int width, int height) {
            x = (int) (Math.random() * width);
            y = (int) (Math.random() * height);
            size = (int) (Math.random() * 4) + 2; // Tamaño entre 2 y 5 píxeles
            speed = (int) (Math.random() * 3) + 1; // Velocidad de caída
        }
        
        public void update(int width, int height) {
            y += speed;
            x -= 1; // Viento suave hacia la izquierda
            
            // Si sale de la pantalla, reaparece arriba
            if (y > height) y = 0;
            if (x < 0) x = width;
        }
    }

    public MenuPanel() {

        setLayout(new BoxLayout(this, BoxLayout.Y_AXIS));
        setBorder(BorderFactory.createEmptyBorder(150, 50, 0, 0));

        java.net.URL imgUrl = getClass().getResource("/Resources/imagenes/fondo_menu_principal4.png");
        if (imgUrl != null) {
            this.backgroundImage = new ImageIcon(imgUrl).getImage();
        } else {
            System.err.println("No se encontró el fondo del menú principal");
        }
        
        JLabel titleLabel = new JLabel("NO WAY BACK");
        try {
            // Cargamos el archivo de la fuente
            java.io.InputStream fontStream = getClass().getResourceAsStream("/Resources/fonts/Nosifer-Regular.ttf");
            if (fontStream != null) {
                Font customFont = Font.createFont(Font.TRUETYPE_FONT, fontStream); //Nosifer-Regular.ttf

                // damos el estilo y el tamaño deseado (el 60f es un float que representa los píxeles)
                Font titleFont = customFont.deriveFont(Font.BOLD, 60f);

                // aplicamos
                titleLabel.setForeground(new Color(149,6 ,6));
                titleLabel.setFont(titleFont);
                titleLabel.setAlignmentX(LEFT_ALIGNMENT);
            } else {
                System.err.println("No se encontró la fuente tipográfica");
            }

        } catch (Exception e) {
            System.out.println("Error al cargar la fuente, usando Arial por defecto.");
            titleLabel.setFont(new Font("Arial", Font.BOLD, 60));
        }   

        playButton = new MenuButton("Jugar");
        optionButton = new MenuButton("Opciones");
        exitButton = new MenuButton("Salir");

        playButton.setAlignmentX(Component.LEFT_ALIGNMENT);
        optionButton.setAlignmentX(Component.LEFT_ALIGNMENT);
        exitButton.setAlignmentX(Component.LEFT_ALIGNMENT);


        add(Box.createVerticalGlue());

        add(titleLabel);
        add(Box.createRigidArea(new Dimension(0, 80)));
        add(playButton);
        add(Box.createRigidArea(new Dimension(0, 20)));
        add(optionButton);
        add(Box.createRigidArea(new Dimension(0, 20)));
        add(exitButton);

        add(Box.createVerticalGlue());

        // Generar 70 partículas iniciales (asumiendo un ancho base de 1920x1080 para esparcirlas)
        for (int i = 0; i < 70; i++) {
            particulas.add(new Ceniza(1920, 1080));
        }

        // Bucle de animación (se actualiza cada 16ms = ~60 FPS)
        particleTimer = new javax.swing.Timer(16, e -> {
            int currentWidth = getWidth() > 0 ? getWidth() : 1920;
            int currentHeight = getHeight() > 0 ? getHeight() : 1080;
            
            for (Ceniza c : particulas) {
                c.update(currentWidth, currentHeight);
            }
            repaint(); // Obliga a redibujar el panel con las nuevas posiciones
        });
        particleTimer.start();

        
    }

    public MenuButton getPlayButton() { return playButton; }
    public MenuButton getOptionButton() { return optionButton; }
    public MenuButton getExitButton() { return exitButton; }


    @Override 
    protected void paintComponent(Graphics g) {
        super.paintComponent(g);

        if (backgroundImage != null) {
            g.drawImage(backgroundImage, 0, 0, getWidth(), getHeight(), this);

            g.setColor(new Color(10, 12, 18, 140));
            g.fillRect(0, 0, getWidth(), getHeight());
        } else {
            g.setColor(Color.BLACK);
            g.fillRect(0, 0, getWidth(), getHeight());
        }

        // Dibujar las partículas de ceniza/nieve
        g.setColor(new Color(200, 200, 200, 100)); // Gris claro semitransparente
        for (Ceniza c : particulas) {
            g.fillOval(c.x, c.y, c.size, c.size);
        }
    }
}

import java.awt.BorderLayout;

import java.awt.Font;

import javax.swing.JLabel;
import javax.swing.JPanel;
import javax.swing.SwingConstants;

public abstract class SplashPanel extends JPanel {

    public SplashPanel(Color backgroundcColor) {
        this.setLayout(new BorderLayout());
        this.setBackground(backgroundcColor);
    }

    protected abstract void initializeComponents();

    protected JLabel createTag(String text, Font font, Color color) {
        JLabel label = new JLabel(text, SwingConstants.CENTER);
        label.setFont(font);
        label.setForeground(color);
        return label;
    }
}
